   207→
   208→export default function TvetApplicationForm() {
   209→  const { addToast, setCurrentPage } = useAppStore();
   210→  const [currentStep, setCurrentStep] = useState(1);
   211→  const [form, setForm] = useState<TvetFormData>(INITIAL_FORM);
   212→  const [errors, setErrors] = useState<Record<string, string>>({});
   213→  const [submitting, setSubmitting] = useState(false);
   214→  const [submitted, setSubmitted] = useState(false);
   215→  const [result, setResult] = useState<{ referenceNumber: string; schoolpayCode: string; programme: string } | null>(null);
   216→
   217→  const updateField = <K extends keyof TvetFormData>(field: K, value: TvetFormData[K]) => {
   218→    setForm((prev) => ({ ...prev, [field]: value }));
   219→    if (errors[field]) setErrors((prev) => { const n = { ...prev }; delete n[field]; return n; });
   220→  };
   221→
   222→  const updateSubject = (section: 'pleSubjects' | 'olevelSubjects' | 'ujtcSubjects', idx: number, key: 'subject' | 'grade', value: string) => {
   223→    setForm((prev) => {
   224→      const arr = [...prev[section]];
   225→      arr[idx] = { ...arr[idx], [key]: value };
   226→      return { ...prev, [section]: arr };
   227→    });
   228→  };
   229→
   230→  const addSubjectRow = (section: 'pleSubjects' | 'olevelSubjects' | 'ujtcSubjects') => {
   231→    setForm((prev) => ({ ...prev, [section]: [...prev[section], { subject: '', grade: '' }] }));
   232→  };
   233→
   234→  const removeSubjectRow = (section: 'pleSubjects' | 'olevelSubjects' | 'ujtcSubjects', idx: number) => {
   235→    setForm((prev) => ({ ...prev, [section]: prev[section].filter((_, i) => i !== idx) }));
   236→  };
   237→
   238→  const updateOtherQual = (idx: number, key: keyof OtherQual, value: string) => {
   239→    setForm((prev) => {
   240→      const arr = [...prev.otherQuals];
   241→      arr[idx] = { ...arr[idx], [key]: value };
   242→      return { ...prev, otherQuals: arr };
   243→    });
   244→  };
   245→
   246→  const updateOtherQualSubject = (qualIdx: number, subIdx: number, key: 'subject' | 'grade', value: string) => {
   247→    setForm((prev) => {
   248→      const arr = [...prev.otherQuals];
   249→      const subs = [...arr[qualIdx].subjects];
   250→      subs[subIdx] = { ...subs[subIdx], [key]: value };
   251→      arr[qualIdx] = { ...arr[qualIdx], subjects: subs };
   252→      return { ...prev, otherQuals: arr };
   253→    });
   254→  };
   255→
   256→  const addOtherQual = () => {
   257→    setForm((prev) => ({
   258→      ...prev,
   259→      otherQuals: [...prev.otherQuals, {
   260→        institution: '', courseName: '', yearSitting: '', regNumber: '', classGrade: '',
   261→        subjects: [{ subject: '', grade: '' }, { subject: '', grade: '' }, { subject: '', grade: '' },
   262→                   { subject: '', grade: '' }, { subject: '', grade: '' }, { subject: '', grade: '' }],
   263→      }],
   264→    }));
   265→  };
   266→
   267→  const removeOtherQual = (idx: number) => {
   268→    setForm((prev) => ({ ...prev, otherQuals: prev.otherQuals.filter((_, i) => i !== idx) }));
   269→  };
   270→
   271→  const updateWorkRecord = (idx: number, key: keyof WorkRecord, value: string) => {
   272→    setForm((prev) => {
   273→      const arr = [...prev.workRecords];
   274→      arr[idx] = { ...arr[idx], [key]: value };
   275→      return { ...prev, workRecords: arr };
   276→    });
   277→  };
   278→
   279→  const addWorkRecord = () => {
   280→    setForm((prev) => ({ ...prev, workRecords: [...prev.workRecords, { organization: '', postHeld: '', period: '' }] }));
   281→  };
   282→
   283→  const removeWorkRecord = (idx: number) => {
   284→    setForm((prev) => ({ ...prev, workRecords: prev.workRecords.filter((_, i) => i !== idx) }));
   285→  };
   286→
   287→  const updateInstChoice = (idx: number, key: keyof InstitutionChoice, value: string) => {
   288→    setForm((prev) => {
   289→      const arr = [...prev.institutionChoices];
   290→      arr[idx] = { ...arr[idx], [key]: value };
   291→      return { ...prev, institutionChoices: arr };
   292→    });
   293→  };
   294→
   295→  /* ──────── validation ──────── */
   296→
   297→  const validateStep = (step: number): boolean => {
   298→    const e: Record<string, string> = {};
   299→    if (step === 1) {
   300→      if (!form.surname.trim()) e.surname = 'Surname is required';
   301→      if (!form.otherNames.trim()) e.otherNames = 'Other names are required';
   302→      if (!form.dob) e.dob = 'Date of birth is required';
   303→      if (!form.sex) e.sex = 'Sex is required';
   304→      if (!form.homeDistrict) e.homeDistrict = 'Home district is required';
   305→      if (!form.county) e.county = 'County is required';
   306→      if (!form.subCounty) e.subCounty = 'Sub-county is required';
   307→      if (!form.religion) e.religion = 'Religious affiliation is required';
   308→      if (!form.email.trim()) e.email = 'Email is required';
   309→      if (form.email && !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(form.email)) e.email = 'Invalid email address';
   310→      if (!form.telephone.trim()) e.telephone = 'Telephone contact is required';
   311→      if (!form.parentGuardianName.trim()) e.parentGuardianName = 'Parent/Guardian name is required';
   312→      if (!form.parentTelephone.trim()) e.parentTelephone = 'Parent telephone is required';
   313→    }
   314→    if (step === 2) {
   315→      if (!form.pleSchoolName.trim()) e.pleSchoolName = 'School name is required';
   316→      if (!form.pleYearSitting) e.pleYearSitting = 'Year is required';
   317→      if (!form.olevelSchoolName.trim()) e.olevelSchoolName = 'School name is required';
   318→      if (!form.olevelYearSitting) e.olevelYearSitting = 'Year is required';
   319→    }
   320→    if (step === 4) {
   321→      if (!form.institutionChoices[0].courseI.trim()) e['choice1courseI'] = 'First choice course is required';
   322→      if (!form.reasonForCourse.trim()) e.reasonForCourse = 'Reason for choosing course is required';
   323→    }
   324→    if (step === 5) {
   325→      if (!form.declarationName.trim()) e.declarationName = 'Declaration name is required';
   326→      if (!form.declarationDate) e.declarationDate = 'Date is required';
   327→    }
   328→    setErrors(e);
   329→    return Object.keys(e).length === 0;
   330→  };
   331→
   332→  const goNext = () => {
   333→    if (validateStep(currentStep)) {
   334→      if (currentStep < 5) setCurrentStep(currentStep + 1);
   335→    } else {
   336→      addToast('Please fill in all required fields', 'error');
   337→    }
   338→  };
   339→
   340→  const goPrev = () => {
   341→    if (currentStep > 1) setCurrentStep(currentStep - 1);
   342→  };
   343→
   344→  /* ──────── submit ──────── */
   345→
   346→  const handleSubmit = async () => {
   347→    if (!validateStep(5)) { addToast('Please fill in the declaration', 'error'); return; }
   348→    setSubmitting(true);
   349→    try {
   350→      const programme = form.institutionChoices[0].courseI || 'TVET Programme';
   351→      const res = await fetch('/api/admissions/tvet', {
   352→        method: 'POST',
   353→        headers: { 'Content-Type': 'application/json' },
   354→        body: JSON.stringify(form),
   355→      });
   356→      const data = await res.json();
   357→      if (data.success) {
   358→        setResult(data.data);
   359→        setSubmitted(true);
   360→        addToast('TVET Application submitted successfully!', 'success');
   361→      } else {
   362→        addToast(data.message || 'Submission failed. Please try again.', 'error');
   363→      }
   364→    } catch {
   365→      addToast('Network error. Please check your connection and try again.', 'error');
   366→    } finally {
   367→      setSubmitting(false);
   368→    }
   369→  };
   370→
   371→  /* ──────── helpers ──────── */
   372→
   373→  const SectionTitle = ({ code, title }: { code: string; title: string }) => (
   374→    <div className="flex items-center gap-3 mb-6">
   375→      <Badge className="text-xs font-bold px-2.5 py-0.5" style={{ background: GOLD, color: PRIMARY }}>{code}</Badge>
   376→      <h3 className="text-lg font-bold text-gray-900">{title}</h3>
   377→    </div>
   378→  );
   379→
   380→  const FieldError = ({ field }: { field: string }) =>
   381→    errors[field] ? <p className="text-xs text-red-500 mt-1">{errors[field]}</p> : null;
   382→
   383→  const RequiredMark = () => <span className="text-red-500 ml-0.5">*</span>;
   384→
   385→  const SubjectGradeTable = ({
   386→    subjects, gradeOptions, section, maxRows = 20,
   387→  }: {
   388→    subjects: SubjectGrade[];
   389→    gradeOptions: string[];
   390→    section: 'pleSubjects' | 'olevelSubjects' | 'ujtcSubjects' | 'otherQualSubjects';
   391→    maxRows?: number;
   392→    qualIdx?: number;
   393→  }) => {
   394→    const handleAdd = () => {
   395→      if (section === 'otherQualSubjects' && qualIdx !== undefined) {
   396→        updateOtherQualSubject(qualIdx, subjects.length, 'subject', '');
   397→      } else {
   398→        addSubjectRow(section as 'pleSubjects' | 'olevelSubjects' | 'ujtcSubjects');
   399→      }
   400→    };
   401→    const handleRemove = (idx: number) => {
   402→      if (section === 'otherQualSubjects' && qualIdx !== undefined) {
   403→        setForm((prev) => {
   404→          const arr = [...prev.otherQuals];
   405→          arr[qualIdx] = { ...arr[qualIdx], subjects: arr[qualIdx].subjects.filter((_, i) => i !== idx) };
   406→          return { ...prev, otherQuals: arr };
   407→        });
   408→      } else {
   409→        removeSubjectRow(section as 'pleSubjects' | 'olevelSubjects' | 'ujtcSubjects', idx);
   410→      }
   411→    };
   412→    const handleUpdate = (idx: number, key: 'subject' | 'grade', value: string) => {
   413→      if (section === 'otherQualSubjects' && qualIdx !== undefined) {
   414→        updateOtherQualSubject(qualIdx, idx, key, value);
   415→      } else {
   416→        updateSubject(section as 'pleSubjects' | 'olevelSubjects' | 'ujtcSubjects', idx, key, value);
   417→      }
   418→    };
   419→
   420→    return (
   421→      <div className="mt-3">
   422→        <div className="rounded-lg border border-gray-200 overflow-hidden">
   423→          <Table>
   424→            <TableHeader>
   425→              <TableRow style={{ background: `${PRIMARY}10` }}>
   426→                <TableHead className="w-10 text-center text-xs font-bold" style={{ color: PRIMARY }}>#</TableHead>
   427→                <TableHead className="text-xs font-bold" style={{ color: PRIMARY }}>Subject</TableHead>
   428→                <TableHead className="w-40 text-xs font-bold" style={{ color: PRIMARY }}>Grade / Score</TableHead>
   429→                <TableHead className="w-10"></TableHead>
   430→              </TableRow>
   431→            </TableHeader>
   432→            <TableBody>
   433→              {subjects.map((row, i) => (
   434→                <TableRow key={i}>
   435→                  <TableCell className="text-center text-xs text-gray-400 font-medium">{i + 1}</TableCell>
   436→                  <TableCell>
   437→                    <Input className="h-8 text-sm" placeholder="Subject name" value={row.subject} onChange={(e) => handleUpdate(i, 'subject', e.target.value)} />
   438→                  </TableCell>
   439→                  <TableCell>
   440→                    <Select value={row.grade} onValueChange={(v) => handleUpdate(i, 'grade', v)}>
   441→                      <SelectTrigger className="h-8 text-sm"><SelectValue placeholder="Grade" /></SelectTrigger>
   442→                      <SelectContent>
   443→                        {gradeOptions.map((g) => (
   444→                          <SelectItem key={g} value={g}>{g}</SelectItem>
   445→                        ))}
   446→                      </SelectContent>
   447→                    </Select>
   448→                  </TableCell>
   449→                  <TableCell>
   450→                    {subjects.length > 1 && (
   451→                      <Button type="button" variant="ghost" size="sm" className="h-7 w-7 p-0 text-red-400 hover:text-red-600 hover:bg-red-50" onClick={() => handleRemove(i)}>
   452→                        <X size={14} />
   453→                      </Button>
   454→                    )}
   455→                  </TableCell>
   456→                </TableRow>
   457→              ))}
   458→            </TableBody>
   459→          </Table>
   460→        </div>
   461→        {subjects.length < maxRows && (
   462→          <Button type="button" variant="outline" size="sm" className="mt-2 h-7 text-xs gap-1" onClick={handleAdd}>
   463→            <Plus size={12} /> Add Subject
   464→          </Button>
   465→        )}
   466→      </div>
   467→    );
   468→  };
   469→
   470→  /* ──────── step renderers ──────── */
   471→
   472→  const renderStep1 = () => (
   473→    <div className="space-y-6">
   474→      <SectionTitle code="SECTION A" title="Particulars of Applicant" />
   475→
   476→      {/* Header with logo info */}
   477→      <div className="rounded-xl border-2 p-4 text-center" style={{ borderColor: `${PRIMARY}20`, background: `${PRIMARY}04` }}>
   478→        <p className="text-xs font-bold uppercase tracking-widest" style={{ color: PRIMARY }}>Republic of Uganda</p>
   479→        <p className="text-sm font-bold mt-1" style={{ color: PRIMARY }}>Ministry of Education and Sports</p>
   480→        <p className="text-xs text-gray-500">TVET Online Application Form</p>
   481→      </div>
   482→
   483→      {/* Names */}
   484→      <div>
   485→        <h4 className="text-sm font-bold text-gray-700 mb-3 flex items-center gap-2"><User size={14} /> Names</h4>
   486→        <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
   487→          <div>
   488→            <Label className="text-sm">Surname <RequiredMark /></Label>
   489→            <Input className="mt-1" placeholder="e.g. Okello" value={form.surname} onChange={(e) => updateField('surname', e.target.value)} />
   490→            <FieldError field="surname" />
   491→          </div>
   492→          <div>
   493→            <Label className="text-sm">Other Names <RequiredMark /></Label>
   494→            <Input className="mt-1" placeholder="e.g. John Peter" value={form.otherNames} onChange={(e) => updateField('otherNames', e.target.value)} />
