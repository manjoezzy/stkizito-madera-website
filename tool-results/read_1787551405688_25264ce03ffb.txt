   102→/* ══════════════════════════════════════════════════════
   103→   ANIMATED COUNTER HOOK
   104→   ══════════════════════════════════════════════════════ */
   105→function useCountUp(target: number, duration = 2000, start = false) {
   106→  const [count, setCount] = useState(0);
   107→  const frameRef = useRef<number>(0);
   108→
   109→  useEffect(() => {
   110→    if (!start) return;
   111→    let startTime: number | null = null;
   112→
   113→    const animate = (timestamp: number) => {
   114→      if (!startTime) startTime = timestamp;
   115→      const progress = Math.min((timestamp - startTime) / duration, 1);
   116→      const eased = 1 - Math.pow(1 - progress, 3); // easeOutCubic
   117→      setCount(Math.floor(eased * target));
   118→      if (progress < 1) {
   119→        frameRef.current = requestAnimationFrame(animate);
   120→      }
   121→    };
   122→
   123→    frameRef.current = requestAnimationFrame(animate);
   124→    return () => cancelAnimationFrame(frameRef.current);
   125→  }, [target, duration, start]);
   126→
   127→  return count;
   128→}
