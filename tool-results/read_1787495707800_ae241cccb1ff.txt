   104→   104→   104→function useCountUp(target: number, duration = 2000, start = false) {
   105→   105→   105→  const [count, setCount] = useState(0);
   106→   106→   106→  const frameRef = useRef<number>(0);
   107→   107→   107→
   108→   108→   108→  useEffect(() => {
   109→   109→   109→    if (!start) return;
   110→   110→   110→    let startTime: number | null = null;
   111→   111→   111→
   112→   112→   112→    const animate = (timestamp: number) => {
   113→   113→   113→      if (!startTime) startTime = timestamp;
   114→   114→   114→      const progress = Math.min((timestamp - startTime) / duration, 1);
   115→   115→   115→      const eased = 1 - Math.pow(1 - progress, 3); // easeOutCubic
   116→   116→   116→      setCount(Math.floor(eased * target));
   117→   117→   117→      if (progress < 1) {
   118→   118→   118→        frameRef.current = requestAnimationFrame(animate);
   119→   119→   119→      }
   120→   120→   120→    };
   121→   121→   121→
   122→   122→   122→    frameRef.current = requestAnimationFrame(animate);
   123→   123→   123→    return () => cancelAnimationFrame(frameRef.current);
   124→   124→   124→  }, [target, duration, start]);
   125→   125→   125→
   126→   126→   126→  return count;
   127→   127→   127→}
