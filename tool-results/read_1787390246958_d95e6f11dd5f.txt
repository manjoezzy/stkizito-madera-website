    50→   109→      const progress = Math.min((timestamp - startTime) / duration, 1);
    51→   110→      const eased = 1 - Math.pow(1 - progress, 3); // easeOutCubic
    52→   111→      setCount(Math.floor(eased * target));
    53→   112→      if (progress < 1) {
    54→   113→        frameRef.current = requestAnimationFrame(animate);
    55→   114→      }
    56→   115→    };
    57→   116→
    58→   117→    frameRef.current = requestAnimationFrame(animate);
    59→   118→    return () => cancelAnimationFrame(frameRef.current);
    60→   119→  }, [target, duration, start]);
    61→   120→
    62→   121→  return count;
    63→   122→}
