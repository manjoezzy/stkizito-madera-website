   150→export default function AdmissionsPage() {

   153→  const [currentStep, setCurrentStep] = useState(1);

   163→    setForm((prev) => ({ ...prev, [field]: value }));

   165→      setErrors((prev) => {

   167→        delete next[field];
   168→        return next;
   169→      });

   171→  };

   174→

   178→    const e: Partial<Record<keyof FormData, string>> = {};
   179→
   180→    if (step === 1) {
   181→      if (!form.fullName.trim()) e.fullName = 'Full name is required';
   182→      if (!form.dob) e.dob = 'Date of birth is required';
   183→      if (!form.gender) e.gender = 'Gender is required';
   184→      if (!form.nationality.trim()) e.nationality = 'Nationality is required';
   185→    }
   186→
   187→    if (step === 2) {
   188→      if (!form.phone.trim()) e.phone = 'Phone number is required';
   189→      else if (!/^\d{10}$/.test(form.phone.replace(/\s/g, ''))) e.phone = 'Enter a valid 10-digit phone number';
   190→      if (!form.email.trim()) e.email = 'Email is required';
   191→      else if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(form.email)) e.email = 'Enter a valid email address';
   192→    }
   193→
   194→    if (step === 3) {
   195→      if (!form.qualification) e.qualification = 'Select your highest qualification';
   196→    }
   197→
   198→    if (step === 4) {
   199→      if (!form.programme) e.programme = 'Select a programme';
   200→      if (!form.intakeYear) e.intakeYear = 'Select intake year';
   201→    }
   202→
   203→    setErrors(e);
   204→    return Object.keys(e).length === 0;
   205→  };

   208→    if (!validateStep(currentStep)) return;

   210→      handleSubmitApplication();

   212→      setCurrentStep((s) => Math.min(s + 1, 5));
   213→    }
   214→  };
   215→
   216→  const goPrev = () => {
   217→    setCurrentStep((s) => Math.max(s - 1, 1));
   218→  };
   219→
   220→  /* ──────── submit application (step 4 → step 5) ──────── */

   226→        method: 'POST',

   234→        setCurrentStep(5);
   235→        addToast('Application submitted successfully!', 'success');

   237→        addToast(data.message || 'Failed to submit application', 'error');
   238→      }
   239→    } catch {
   240→      addToast('Network error. Please try again.', 'error');
   241→    } finally {
   242→      setSubmitting(false);
   243→    }

   259→          phone: form.phone,
   260→          email: form.email,
   261→          amount: selectedProgrammeFee,
   262→          programme: form.programme,
   263→          intakeYear: form.intakeYear,
   264→        }),
   265→      });

   268→      if (data.success) {
   269→        setTransactionRef(data.data.transactionRef);
   270→        setPaymentStatus('dialog');

   272→        setPaymentError(data.message || 'Payment initiation failed');
