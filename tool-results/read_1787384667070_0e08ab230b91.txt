   147→/* ══════════════════════════════════════════════════════
   148→   ANIMATED COUNTER HOOK
   149→   ══════════════════════════════════════════════════════ */
   150→function useCountUp(target: number, duration = 2000, start = false) {
   151→  const [count, setCount] = useState(0);
   152→  const frameRef = useRef<number>(0);
   153→
   154→  useEffect(() => {
   155→    if (!start) return;
   156→    let startTime: number | null = null;
   157→
   158→    const animate = (timestamp: number) => {
   159→      if (!startTime) startTime = timestamp;
   160→      const progress = Math.min((timestamp - startTime) / duration, 1);
   161→      const eased = 1 - Math.pow(1 - progress, 3); // easeOutCubic
   162→      setCount(Math.floor(eased * target));
   163→      if (progress < 1) {
   164→        frameRef.current = requestAnimationFrame(animate);
   165→      }
   166→    };
   167→
   168→    frameRef.current = requestAnimationFrame(animate);
   169→    return () => cancelAnimationFrame(frameRef.current);
   170→  }, [target, duration, start]);
   171→
   172→  return count;
   173→}
