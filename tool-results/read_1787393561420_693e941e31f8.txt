   100→   100→   ANIMATED COUNTER HOOK
   101→   101→   ══════════════════════════════════════════════════════ */
   102→   102→function useCountUp(target: number, duration = 2000, start = false) {
   103→   103→  const [count, setCount] = useState(0);
   104→   104→  const frameRef = useRef<number>(0);
   105→   105→
   106→   106→  useEffect(() => {
   107→   107→    if (!start) return;
   108→   108→    let startTime: number | null = null;
   109→   109→
   110→   110→    const animate = (timestamp: number) => {
   111→   111→      if (!startTime) startTime = timestamp;
   112→   112→      const progress = Math.min((timestamp - startTime) / duration, 1);
   113→   113→      const eased = 1 - Math.pow(1 - progress, 3); // easeOutCubic
   114→   114→      setCount(Math.floor(eased * target));
   115→   115→      if (progress < 1) {
   116→   116→        frameRef.current = requestAnimationFrame(animate);
   117→   117→      }
   118→   118→    };
   119→   119→
   120→   120→    frameRef.current = requestAnimationFrame(animate);
   121→   121→    return () => cancelAnimationFrame(frameRef.current);
   122→   122→  }, [target, duration, start]);
   123→   123→
   124→   124→  return count;
   125→   125→}
