   148→/* ══════════════════════════════════════════════════════
   149→   ANIMATED COUNTER HOOK
   150→   ══════════════════════════════════════════════════════ */
   151→function useCountUp(target: number, duration = 2000, start = false) {
   152→  const [count, setCount] = useState(0);
   153→  const frameRef = useRef<number>(0);
   154→
   155→  useEffect(() => {
   156→    if (!start) return;
   157→    let startTime: number | null = null;
   158→
   159→    const animate = (timestamp: number) => {
   160→      if (!startTime) startTime = timestamp;
   161→      const progress = Math.min((timestamp - startTime) / duration, 1);
   162→      const eased = 1 - Math.pow(1 - progress, 3); // easeOutCubic
   163→      setCount(Math.floor(eased * target));
   164→      if (progress < 1) {
   165→        frameRef.current = requestAnimationFrame(animate);
   166→      }
   167→    };
   168→
   169→    frameRef.current = requestAnimationFrame(animate);
   170→    return () => cancelAnimationFrame(frameRef.current);
   171→  }, [target, duration, start]);
   172→
   173→  return count;
   174→}
