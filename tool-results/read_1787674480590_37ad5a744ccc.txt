   128→   128→function computeProgrammeSummary(alumni: Alumni[]): Record<string, number> {
   129→   129→  const counts: Record<string, number> = {};
   130→   130→  for (const a of alumni) {
   131→   131→    const p = a.programme || 'Unspecified';
   132→   132→    counts[p] = (counts[p] || 0) + 1;
   133→   133→  }
   134→   134→  return counts;
   135→   135→}
