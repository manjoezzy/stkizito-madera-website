   153→
   154→export default function AdmissionsPage() {
   155→  const { addToast, setCurrentPage } = useAppStore();
   156→
   157→  const [currentStep, setCurrentStep] = useState(1);
   158→  const [form, setForm] = useState<FormData>(INITIAL_FORM);

   167→    setForm((prev) => ({ ...prev, [field]: value }));

   169→      setErrors((prev) => {
   170→        const next = { ...prev };
   171→        delete next[field];
   172→        return next;
   173→      });
   174→    }
   175→  };
   176→
   177→  const selectedProgrammeFee = form.programme ? getProgrammeFee(form.programme) : 0;
   178→

   180→

   182→    const e: Partial<Record<keyof FormData, string>> = {};
   183→
   184→    if (step === 1) {
   185→      if (!form.fullName.trim()) e.fullName = 'Full name is required';
   186→      if (!form.dob) e.dob = 'Date of birth is required';
   187→      if (!form.gender) e.gender = 'Gender is required';
   188→      if (!form.nationality.trim()) e.nationality = 'Nationality is required';
   189→    }
   190→
   191→    if (step === 2) {
   192→      if (!form.phone.trim()) e.phone = 'Phone number is required';
   193→      else if (!/^\d{10}$/.test(form.phone.replace(/\s/g, ''))) e.phone = 'Enter a valid 10-digit phone number';
   194→      if (!form.email.trim()) e.email = 'Email is required';
   195→      else if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(form.email)) e.email = 'Enter a valid email address';
   196→    }
   197→
   198→    if (step === 3) {
   199→      if (!form.qualification) e.qualification = 'Select your highest qualification';
   200→    }
   201→
   202→    if (step === 4) {
   203→      if (!form.programme) e.programme = 'Select a programme';
   204→      if (!form.intakeYear) e.intakeYear = 'Select intake year';
   205→    }
   206→
   207→    setErrors(e);
   208→    return Object.keys(e).length === 0;
   209→  };

   212→    if (!validateStep(currentStep)) return;
   213→    if (currentStep === 4) {
   214→      handleSubmitApplication();
   215→    } else {
   216→      setCurrentStep((s) => Math.min(s + 1, 5));
   217→    }
   218→  };
   219→
   220→  const goPrev = () => {
   221→    setCurrentStep((s) => Math.max(s - 1, 1));
   222→  };
   223→
   224→  /* ──────── submit application (step 4 → step 5) ──────── */

   226→  const handleSubmitApplication = async () => {

   229→      const res = await fetch('/api/admissions', {
   230→        method: 'POST',
   231→        headers: { 'Content-Type': 'application/json' },
   232→        body: JSON.stringify(form),
   233→      });

   235→
   236→      if (data.success) {
   237→        setReferenceNumber(data.data.referenceNumber);
   238→        setCurrentStep(5);
   239→        addToast('Application submitted successfully!', 'success');
   240→      } else {
   241→        addToast(data.message || 'Failed to submit application', 'error');
   242→      }
   243→    } catch {
   244→      addToast('Network error. Please try again.', 'error');
   245→    } finally {
   246→      setSubmitting(false);
   247→    }
   248→  };

   250→  /* ──────── SchoolPay payment ──────── */

   257→        method: 'POST',
   258→        headers: { 'Content-Type': 'application/json' },

   260→          action: 'initiate',
   261→          referenceNumber,
   262→          fullName: form.fullName,
   263→          phone: form.phone,
   264→          email: form.email,
   265→          amount: selectedProgrammeFee,
   266→          programme: form.programme,
   267→          intakeYear: form.intakeYear,
   268→        }),
   269→      });
   270→      const data = await res.json();
   271→
   272→      if (data.success) {
   273→        setTransactionRef(data.data.transactionRef);
   274→        setPaymentStatus('dialog');
   275→      } else {
   276→        setPaymentError(data.message || 'Payment initiation failed');
   277→        setPaymentStatus('idle');

   295→        }),
