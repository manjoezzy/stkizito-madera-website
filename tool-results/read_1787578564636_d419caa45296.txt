     1→'use client';
     2→
     3→import { useState } from 'react';
     4→import { motion, AnimatePresence } from 'framer-motion';
     5→import {
     6→  Search,
     7→  ClipboardList,
     8→  CheckCircle,
     9→  XCircle,
    10→  Clock,
    11→  Loader2,
    12→  Copy,
    13→  FileText,
    14→  User,
    15→  BookOpen,
    16→  CreditCard,
    17→  Phone,
    18→  Mail,
    19→  Calendar,
    20→  ArrowLeft,
    21→  GraduationCap,
    22→  ShieldCheck,
    23→  Eye,
    24→  CircleCheckBig,
    25→} from 'lucide-react';
    26→import { Button } from '@/components/ui/button';
    27→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';
    28→import { Input } from '@/components/ui/input';
    29→import { Label } from '@/components/ui/label';
    30→import { Badge } from '@/components/ui/badge';
    31→import { useAppStore } from '@/store/useAppStore';
    32→import { formatCurrency } from '@/lib/schoolpay';
    33→
    34→const PRIMARY = '#1a3a6b';
    35→const PRIMARY_LIGHT = '#2756a0';
    36→const GOLD = '#f5c518';
    37→
    38→interface ApplicationData {
    39→  referenceNumber: string;
    40→  fullName: string;
    41→  programme: string | null;
    42→  intakeYear: string | null;
    43→  status: string;
    44→  schoolpayCode: string | null;
    45→  paymentStatus: string;
    46→  paymentAmount: number | null;
    47→  createdAt: string;
    48→  phone: string;
    49→  email: string;
    50→  documents: Array<{
    51→    id: string;
    52→    fileName: string;
    53→    documentType: string;
    54→    createdAt: string;
    55→  }>;
    56→  grades: string | null;
    57→}
    58→
    59→const STATUS_STEPS = [
    60→  { key: 'applied', label: 'Applied' },
    61→  { key: 'documents_verified', label: 'Documents Verified' },
    62→  { key: 'under_review', label: 'Under Review' },
    63→  { key: 'decision', label: 'Approved / Rejected' },
    64→  { key: 'enrolled', label: 'Enrolled' },
    65→];
    66→
    67→function getStatusStepIndex(status: string): number {
    68→  switch (status) {
    69→    case 'pending': return 0;
    70→    case 'approved': return 3;
    71→    case 'rejected': return 3;
    72→    case 'enrolled': return 4;
    73→    default: return 0;
    74→  }
    75→}
    76→
    77→function getStatusConfig(status: string) {
    78→  switch (status) {
    79→    case 'pending':
    80→      return { label: 'Pending', color: 'bg-amber-100 text-amber-800 border-amber-200', dot: 'bg-amber-500' };
    81→    case 'approved':
    82→      return { label: 'Approved', color: 'bg-emerald-100 text-emerald-800 border-emerald-200', dot: 'bg-emerald-500' };
    83→    case 'rejected':
    84→      return { label: 'Rejected', color: 'bg-red-100 text-red-800 border-red-200', dot: 'bg-red-500' };
    85→    case 'enrolled':
    86→      return { label: 'Enrolled', color: 'bg-blue-100 text-blue-800 border-blue-200', dot: 'bg-blue-500' };
    87→    default:
    88→      return { label: 'Unknown', color: 'bg-gray-100 text-gray-800 border-gray-200', dot: 'bg-gray-500' };
    89→  }
    90→}
    91→
    92→export default function TrackApplicationPage() {
    93→  const { addToast, setCurrentPage } = useAppStore();
    94→
    95→  const [searchInput, setSearchInput] = useState('');
    96→  const [searching, setSearching] = useState(false);
    97→  const [application, setApplication] = useState<ApplicationData | null>(null);
    98→  const [notFound, setNotFound] = useState(false);
    99→
   100→  const handleSearch = async () => {
   101→    const query = searchInput.trim();
   102→    if (!query) {
   103→      addToast('Enter a reference number or phone number', 'error');
   104→      return;
   105→    }
   106→
   107→    setSearching(true);
   108→    setNotFound(false);
   109→    setApplication(null);
   110→
   111→    try {
   112→      const isRef = query.startsWith('SKT-');
   113→      const url = isRef
   114→        ? `/api/admissions?ref=${encodeURIComponent(query)}`
   115→        : `/api/admissions?search=${encodeURIComponent(query)}`;
   116→
   117→      const res = await fetch(url);
   118→      const data = await res.json();
   119→
   120→      if (data.success && data.data) {
   121→        const app = Array.isArray(data.data) ? data.data[0] : data.data;
   122→        if (app) {
   123→          setApplication(app);
   124→        } else {
   125→          setNotFound(true);
   126→        }
   127→      } else {
   128→        setNotFound(true);
   129→      }
   130→    } catch {
   131→      addToast('Network error. Please try again.', 'error');
   132→    } finally {
   133→      setSearching(false);
   134→    }
   135→  };
   136→
   137→  const handleKeyDown = (e: React.KeyboardEvent) => {
   138→    if (e.key === 'Enter') handleSearch();
   139→  };
   140→
   141→  const copyToClipboard = (text: string) => {
   142→    navigator.clipboard.writeText(text).then(() => {
   143→      addToast('Copied to clipboard!', 'success');
   144→    }).catch(() => {
   145→      addToast('Failed to copy', 'error');
   146→    });
   147→  };
   148→
   149→  const statusConfig = application ? getStatusConfig(application.status) : null;
   150→  const currentStepIdx = application ? getStatusStepIndex(application.status) : -1;
   151→  const parsedGrades = application?.grades
   152→    ? (() => { try { return JSON.parse(application.grades); } catch { return null; } })()
   153→    : null;
   154→
   155→  return (
   156→    <div className="min-h-screen flex flex-col">
   157→      {/* Header */}
   158→      <header
   159→        className="pt-[104px] lg:pt-[108px] pb-16 relative overflow-hidden"
   160→        style={{ background: `linear-gradient(135deg, ${PRIMARY} 0%, ${PRIMARY_LIGHT} 50%, ${PRIMARY} 100%)` }}
   161→      >
   162→        <div className="absolute inset-0 opacity-10">
   163→          <div className="absolute top-10 left-10 size-64 rounded-full" style={{ background: GOLD, filter: 'blur(80px)' }} />
   164→          <div className="absolute bottom-10 right-10 size-48 rounded-full" style={{ background: GOLD, filter: 'blur(60px)' }} />
   165→        </div>
   166→        <div className="max-w-4xl mx-auto px-4 sm:px-6 text-center relative z-10">
   167→          <motion.div
   168→            initial={{ opacity: 0, y: 20 }}
   169→            animate={{ opacity: 1, y: 0 }}
   170→            transition={{ duration: 0.6 }}
   171→          >
   172→            <Badge
   173→              className="mb-4 text-xs font-semibold px-4 py-1.5"
   174→              style={{ background: GOLD, color: PRIMARY, borderColor: 'transparent' }}
   175→            >
   176→              <Eye className="size-3.5 mr-1.5" />
   177→              Application Tracking
   178→            </Badge>
   179→            <h1 className="text-4xl sm:text-5xl font-extrabold text-white tracking-tight">
   180→              Track Your Application
   181→            </h1>
   182→            <p className="mt-4 text-lg text-blue-100 max-w-2xl mx-auto">
   183→              Enter your application reference number or phone number to check the status of your admission.
   184→            </p>
   185→          </motion.div>
   186→        </div>
   187→      </header>
   188→
   189→      <main className="flex-1 py-12 px-4">
   190→        <div className="max-w-3xl mx-auto space-y-8">
   191→          {/* Search Card */}
   192→          <motion.div
   193→            initial={{ opacity: 0, y: 20 }}
   194→            animate={{ opacity: 1, y: 0 }}
   195→            transition={{ duration: 0.4, delay: 0.1 }}
   196→          >
   197→            <Card className="shadow-lg border-gray-100">
   198→              <CardContent className="p-6">
   199→                <div className="flex flex-col sm:flex-row gap-3">
   200→                  <div className="flex-1">
   201→                    <Label htmlFor="search-input" className="text-sm font-medium text-gray-700 mb-1.5 block">
   202→                      Reference Number or Phone Number
   203→                    </Label>
   204→                    <Input
   205→                      id="search-input"
   206→                      type="text"
   207→                      placeholder="e.g. SKT-2025-12345 or 0771234567"
   208→                      value={searchInput}
   209→                      onChange={(e) => setSearchInput(e.target.value)}
   210→                      onKeyDown={handleKeyDown}
   211→                      className="h-11"
   212→                    />
   213→                  </div>
   214→                  <div className="flex items-end">
   215→                    <Button
   216→                      onClick={handleSearch}
   217→                      disabled={searching}
   218→                      className="h-11 px-6 cursor-pointer"
   219→                      style={{ background: PRIMARY }}
   220→                    >
   221→                      {searching ? (
   222→                        <>
   223→                          <Loader2 className="size-4 mr-2 animate-spin" />
   224→                          Searching...
   225→                        </>
   226→                      ) : (
   227→                        <>
   228→                          <Search className="size-4 mr-2" />
   229→                          Track Application
   230→                        </>
   231→                      )}
   232→                    </Button>
   233→                  </div>
   234→                </div>
   235→              </CardContent>
   236→            </Card>
   237→          </motion.div>
   238→
   239→          {/* Not Found */}
   240→          <AnimatePresence>
   241→            {notFound && (
   242→              <motion.div
   243→                initial={{ opacity: 0, y: 20 }}
   244→                animate={{ opacity: 1, y: 0 }}
   245→                exit={{ opacity: 0, y: -20 }}
   246→              >
   247→                <Card className="border-red-200 bg-red-50">
   248→                  <CardContent className="p-8 text-center">
   249→                    <div className="mx-auto size-16 rounded-full bg-red-100 flex items-center justify-center mb-4">
   250→                      <XCircle className="size-8 text-red-400" />
   251→                    </div>
   252→                    <h3 className="text-lg font-bold text-gray-900">Application Not Found</h3>
   253→                    <p className="text-sm text-gray-500 mt-2 max-w-md mx-auto">
   254→                      We couldn't find an application matching your search. Please check your reference number or phone number and try again.
   255→                    </p>
   256→                    <Button
   257→                      onClick={() => setCurrentPage('admissions')}
   258→                      variant="outline"
   259→                      className="mt-4 cursor-pointer"
   260→                    >
   261→                      <ArrowLeft className="size-4 mr-2" />
   262→                      Apply Now
   263→                    </Button>
   264→                  </CardContent>
   265→                </Card>
   266→              </motion.div>
   267→            )}
   268→          </AnimatePresence>
   269→
   270→          {/* Application Dashboard */}
   271→          <AnimatePresence>
   272→            {application && statusConfig && (
   273→              <motion.div
   274→                initial={{ opacity: 0, y: 20 }}
   275→                animate={{ opacity: 1, y: 0 }}
   276→                exit={{ opacity: 0, y: -20 }}
   277→                className="space-y-6"
   278→              >
   279→                {/* Reference Number & Status */}
   280→                <Card className="shadow-lg border-gray-100">
   281→                  <CardContent className="p-6">
   282→                    <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   283→                      <div>
   284→                        <p className="text-xs text-gray-400 uppercase tracking-wider font-semibold">Reference Number</p>
   285→                        <div className="flex items-center gap-2 mt-1">
   286→                          <p className="text-2xl sm:text-3xl font-extrabold font-mono" style={{ color: PRIMARY }}>
   287→                            {application.referenceNumber}
   288→                          </p>
   289→                          <button
   290→                            onClick={() => copyToClipboard(application.referenceNumber)}
   291→                            className="p-1.5 rounded-lg hover:bg-gray-100 transition-colors cursor-pointer"
   292→                            title="Copy reference"
   293→                          >
   294→                            <Copy className="size-4 text-gray-400" />
   295→                          </button>
   296→                        </div>
   297→                      </div>
   298→                      <Badge className={`${statusConfig.color} border text-sm px-4 py-1.5 font-semibold`}>
   299→                        <span className={`size-2 rounded-full mr-2 ${statusConfig.dot}`} />
   300→                        {statusConfig.label}
   301→                      </Badge>
   302→                    </div>
   303→                  </CardContent>
   304→                </Card>
   305→
   306→                {/* Status Timeline */}
   307→                <Card className="shadow-lg border-gray-100">
   308→                  <CardHeader className="pb-4">
   309→                    <CardTitle className="text-base font-bold text-gray-900 flex items-center gap-2">
   310→                      <Clock className="size-4" style={{ color: PRIMARY }} />
   311→                      Application Progress
   312→                    </CardTitle>
   313→                  </CardHeader>
   314→                  <CardContent className="pb-6">
   315→                    <div className="flex items-center gap-0 overflow-x-auto pb-2">
   316→                      {STATUS_STEPS.map((step, idx) => {
   317→                        const isCompleted = idx < currentStepIdx;
   318→                        const isCurrent = idx === currentStepIdx;
   319→                        const isRejected = application.status === 'rejected' && idx === 3;
   320→
   321→                        return (
   322→                          <div key={step.key} className="flex items-center">
   323→                            <div className="flex flex-col items-center min-w-[80px]">
   324→                              <div
   325→                                className={`size-10 rounded-full flex items-center justify-center border-2 transition-all ${
   326→                                  isCompleted
   327→                                    ? 'border-green-400 bg-green-50'
   328→                                    : isCurrent && isRejected
   329→                                    ? 'border-red-400 bg-red-50'
   330→                                    : isCurrent
   331→                                    ? 'border-amber-400 bg-amber-50'
   332→                                    : 'border-gray-200 bg-gray-50'
   333→                                }`}
   334→                              >
   335→                                {isCompleted ? (
   336→                                  <CheckCircle className="size-5 text-green-500" />
   337→                                ) : isCurrent && isRejected ? (
   338→                                  <XCircle className="size-5 text-red-500" />
   339→                                ) : isCurrent ? (
   340→                                  <div className={`size-3 rounded-full animate-pulse ${isRejected ? 'bg-red-400' : 'bg-amber-400'}`} />
   341→                                ) : (
   342→                                  <div className="size-2 rounded-full bg-gray-300" />
   343→                                )}
   344→                              </div>
   345→                              <span
   346→                                className={`text-xs mt-2 text-center font-medium ${
   347→                                  isCompleted
   348→                                    ? 'text-green-700'
   349→                                    : isCurrent
   350→                                    ? isRejected ? 'text-red-700' : 'text-amber-700'
   351→                                    : 'text-gray-400'
   352→                                }`}
   353→                              >
   354→                                {step.label}
   355→                              </span>
   356→                            </div>
   357→                            {idx < STATUS_STEPS.length - 1 && (
   358→                              <div
   359→                                className={`h-0.5 w-8 sm:w-12 mx-1 mb-5 rounded transition-colors ${
   360→                                  idx < currentStepIdx ? 'bg-green-400' : 'bg-gray-200'
   361→                                }`}
   362→                              />
   363→                            )}
   364→                          </div>
   365→                        );
   366→                      })}
   367→                    </div>
   368→                  </CardContent>
   369→                </Card>
   370→
   371→                {/* Personal Info Summary */}
   372→                <Card className="shadow-lg border-gray-100">
   373→                  <CardHeader className="pb-4">
   374→                    <CardTitle className="text-base font-bold text-gray-900 flex items-center gap-2">
   375→                      <User className="size-4" style={{ color: PRIMARY }} />
   376→                      Applicant Information
   377→                    </CardTitle>
   378→                  </CardHeader>
   379→                  <CardContent className="pb-6">
   380→                    <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   381→                      <div className="flex items-start gap-3">
   382→                        <User className="size-4 text-gray-400 mt-0.5 shrink-0" />
   383→                        <div>
   384→                          <p className="text-xs text-gray-400">Full Name</p>
   385→                          <p className="text-sm font-semibold text-gray-900">{application.fullName}</p>
   386→                        </div>
   387→                      </div>
   388→                      <div className="flex items-start gap-3">
   389→                        <BookOpen className="size-4 text-gray-400 mt-0.5 shrink-0" />
   390→                        <div>
   391→                          <p className="text-xs text-gray-400">Programme</p>
   392→                          <p className="text-sm font-semibold text-gray-900">{application.programme || 'N/A'}</p>
   393→                        </div>
   394→                      </div>
   395→                      <div className="flex items-start gap-3">
   396→                        <Calendar className="size-4 text-gray-400 mt-0.5 shrink-0" />
   397→                        <div>
   398→                          <p className="text-xs text-gray-400">Date Applied</p>
   399→                          <p className="text-sm font-semibold text-gray-900">
   400→                            {new Date(application.createdAt).toLocaleDateString('en-GB', {
   401→                              day: '2-digit',
   402→                              month: 'long',
   403→                              year: 'numeric',
   404→                            })}
   405→                          </p>
   406→                        </div>
   407→                      </div>
   408→                      <div className="flex items-start gap-3">
   409→                        <GraduationCap className="size-4 text-gray-400 mt-0.5 shrink-0" />
   410→                        <div>
   411→                          <p className="text-xs text-gray-400">Intake Year</p>
   412→                          <p className="text-sm font-semibold text-gray-900">{application.intakeYear || 'N/A'}</p>
   413→                        </div>
   414→                      </div>
   415→                      <div className="flex items-start gap-3">
   416→                        <Phone className="size-4 text-gray-400 mt-0.5 shrink-0" />
   417→                        <div>
   418→                          <p className="text-xs text-gray-400">Phone</p>
   419→                          <p className="text-sm font-semibold text-gray-900">{application.phone}</p>
   420→                        </div>
   421→                      </div>
   422→                      <div className="flex items-start gap-3">
   423→                        <Mail className="size-4 text-gray-400 mt-0.5 shrink-0" />
   424→                        <div>
   425→                          <p className="text-xs text-gray-400">Email</p>
   426→                          <p className="text-sm font-semibold text-gray-900">{application.email}</p>
   427→                        </div>
   428→                      </div>
   429→                    </div>
   430→                  </CardContent>
   431→                </Card>
   432→
   433→                {/* SchoolPay Code & Payment */}
   434→                {(application.schoolpayCode || application.paymentStatus) && (
   435→                  <Card className="shadow-lg border-gray-100">
   436→                    <CardHeader className="pb-4">
   437→                      <CardTitle className="text-base font-bold text-gray-900 flex items-center gap-2">
   438→                        <CreditCard className="size-4" style={{ color: PRIMARY }} />
   439→                        Payment Information
   440→                      </CardTitle>
   441→                    </CardHeader>
   442→                    <CardContent className="pb-6 space-y-4">
   443→                      {application.schoolpayCode && (
   444→                        <div className="flex items-center justify-between p-4 rounded-lg bg-amber-50 border border-amber-200">
   445→                          <div>
   446→                            <p className="text-xs text-amber-600 font-medium">SchoolPay Code</p>
   447→                            <p className="text-xl font-extrabold font-mono tracking-wider" style={{ color: PRIMARY }}>
   448→                              {application.schoolpayCode}
   449→                            </p>
   450→                          </div>
   451→                          <button
   452→                            onClick={() => copyToClipboard(application.schoolpayCode!)}
   453→                            className="p-2 rounded-lg hover:bg-amber-100 transition-colors cursor-pointer"
   454→                            title="Copy code"
   455→                          >
   456→                            <Copy className="size-5 text-amber-600" />
   457→                          </button>
   458→                        </div>
   459→                      )}
   460→
   461→                      {application.paymentStatus && (
   462→                        <div className="flex items-center justify-between py-2 border-t border-gray-100">
   463→                          <span className="text-sm text-gray-500">Payment Status</span>
   464→                          <Badge
   465→                            className={`${
   466→                              application.paymentStatus === 'paid'
   467→                                ? 'bg-emerald-100 text-emerald-800 border-emerald-200'
   468→                                : application.paymentStatus === 'failed'
   469→                                ? 'bg-red-100 text-red-800 border-red-200'

   471→                            } border text-xs font-semibold`}
   472→                          >
   473→                            {application.paymentStatus.charAt(0).toUpperCase() + application.paymentStatus.slice(1)}
   474→                          </Badge>
   475→                        </div>
   476→                      )}
   477→
   478→                      {application.paymentAmount != null && application.paymentAmount > 0 && (
   479→                        <div className="flex items-center justify-between py-2 border-t border-gray-100">
   480→                          <span className="text-sm text-gray-500">Amount Paid</span>
   481→                          <span className="text-sm font-bold" style={{ color: GOLD }}>
   482→                            {formatCurrency(application.paymentAmount)}
   483→                          </span>
   484→                        </div>
   485→                      )}
   486→
   487→                      {application.schoolpayCode && application.paymentStatus !== 'paid' && (
   488→                        <div className="rounded-lg bg-blue-50 border border-blue-200 p-3">
   489→                          <p className="text-xs text-blue-700">
   490→                            <ShieldCheck className="size-3.5 inline-block mr-1" />
   491→                            Dial <span className="font-bold font-mono">*210#</span> on MTN or <span className="font-bold font-mono">*185#</span> on Airtel, enter the code above, and follow prompts.
   492→                          </p>
   493→                        </div>
   494→                      )}
   495→                    </CardContent>
   496→                  </Card>
   497→                )}
   498→
   499→                {/* Grades */}
   500→                {parsedGrades && Array.isArray(parsedGrades) && parsedGrades.length > 0 && (
   501→                  <Card className="shadow-lg border-gray-100">
   502→                    <CardHeader className="pb-4">
   503→                      <CardTitle className="text-base font-bold text-gray-900 flex items-center gap-2">
   504→                        <ClipboardList className="size-4" style={{ color: PRIMARY }} />
   505→                        Academic Grades

   507→                    </CardHeader>

   509→                      <div className="rounded-lg border border-gray-200 overflow-hidden">

   513→                        </div>

   515→                          {parsedGrades.map((g: { subject: string; grade: string }, idx: number) => (
   516→                            <div key={idx} className="grid grid-cols-2 gap-0 px-4 py-2 border-b border-gray-100 last:border-b-0">
   517→                              <span className="text-sm text-gray-700">{g.subject}</span>

   521→                        </div>
   522→                      </div>
   523→                    </CardContent>
   524→                  </Card>
   525→                )}
   526→
   527→                {/* Uploaded Documents */}
   528→                {application.documents && application.documents.length > 0 && (
