   198→   198→      if (!form.fullName.trim()) e.fullName = 'Full name is required';

   202→   202→    }
   203→   203→

   205→   205→      if (!form.phone.trim()) e.phone = 'Phone number is required';

   213→   213→    }

   215→   215→    if (step === 4) {
   216→   216→      if (!form.programme) e.programme = 'Select a programme';

   218→   218→    }

   230→   230→    }
