   103→   103→   ANIMATED COUNTER HOOK
   104→   104→   ══════════════════════════════════════════════════════ */
   105→   105→function useCountUp(target: number, duration = 2000, start = false) {
   106→   106→  const [count, setCount] = useState(0);
   107→   107→  const frameRef = useRef<number>(0);
   108→   108→
   109→   109→  useEffect(() => {
   110→   110→    if (!start) return;
   111→   111→    let startTime: number | null = null;
   112→   112→
   113→   113→    const animate = (timestamp: number) => {
   114→   114→      if (!startTime) startTime = timestamp;
   115→   115→      const progress = Math.min((timestamp - startTime) / duration, 1);
   116→   116→      const eased = 1 - Math.pow(1 - progress, 3); // easeOutCubic
   117→   117→      setCount(Math.floor(eased * target));
   118→   118→      if (progress < 1) {
   119→   119→        frameRef.current = requestAnimationFrame(animate);
   120→   120→      }
   121→   121→    };
   122→   122→
   123→   123→    frameRef.current = requestAnimationFrame(animate);
   124→   124→    return () => cancelAnimationFrame(frameRef.current);
   125→   125→  }, [target, duration, start]);
   126→   126→
   127→   127→  return count;
   128→   128→}
