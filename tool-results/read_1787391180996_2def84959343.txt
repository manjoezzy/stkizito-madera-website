    99→    99→function useCountUp(target: number, duration = 2000, start = false) {
   100→   100→  const [count, setCount] = useState(0);
   101→   101→  const frameRef = useRef<number>(0);
   102→   102→
   103→   103→  useEffect(() => {
   104→   104→    if (!start) return;
   105→   105→    let startTime: number | null = null;
   106→   106→
   107→   107→    const animate = (timestamp: number) => {
   108→   108→      if (!startTime) startTime = timestamp;
   109→   109→      const progress = Math.min((timestamp - startTime) / duration, 1);
   110→   110→      const eased = 1 - Math.pow(1 - progress, 3); // easeOutCubic
   111→   111→      setCount(Math.floor(eased * target));
   112→   112→      if (progress < 1) {
   113→   113→        frameRef.current = requestAnimationFrame(animate);
   114→   114→      }
   115→   115→    };
   116→   116→
   117→   117→    frameRef.current = requestAnimationFrame(animate);
   118→   118→    return () => cancelAnimationFrame(frameRef.current);
   119→   119→  }, [target, duration, start]);
   120→   120→
   121→   121→  return count;
   122→   122→}
