   216→export default function TvetApplicationForm() {
   217→  const { addToast, setCurrentPage } = useAppStore();
   218→  const [currentStep, setCurrentStep] = useState(1);
   219→  const [form, setForm] = useState<TvetFormData>(INITIAL_FORM);
   220→  const [errors, setErrors] = useState<Record<string, string>>({});
   221→  const [submitting, setSubmitting] = useState(false);
   222→  const [submitted, setSubmitted] = useState(false);
   223→  const [result, setResult] = useState<{ referenceNumber: string; schoolpayCode: string; programme: string } | null>(null);
   224→  const photoInputRef = useRef<HTMLInputElement>(null);
   225→  const [photoDragOver, setPhotoDragOver] = useState(false);
   226→
   227→  /* ──────── passport photo handling ──────── */
   228→
   229→  const handlePhotoSelect = useCallback((file: File | null) => {
   230→    if (!file) return;
   231→    const allowedTypes = ['image/jpeg', 'image/png', 'image/webp'];
   232→    if (!allowedTypes.includes(file.type)) {
   233→      addToast('Invalid file type. Please upload a JPEG, PNG, or WebP image only.', 'error');
   234→      return;
   235→    }
   236→    if (file.size > 1.5 * 1024 * 1024) {
   237→      addToast('File too large. Maximum size is 1.5MB.', 'error');
   238→      return;
   239→    }
   240→    const preview = URL.createObjectURL(file);
   241→    setForm((prev) => ({ ...prev, passportPhoto: file, passportPhotoPreview: preview }));
   242→    if (errors.passportPhoto) setErrors((prev) => { const n = { ...prev }; delete n.passportPhoto; return n; });
   243→  }, [addToast, errors]);
   244→
   245→  const removePhoto = useCallback(() => {
   246→    if (form.passportPhotoPreview) URL.revokeObjectURL(form.passportPhotoPreview);
   247→    setForm((prev) => ({ ...prev, passportPhoto: null, passportPhotoPreview: '' }));
   248→    if (photoInputRef.current) photoInputRef.current.value = '';
   249→  }, [form.passportPhotoPreview]);
   250→
   251→  const handlePhotoDrop = useCallback((e: React.DragEvent) => {
   252→    e.preventDefault();
   253→    e.stopPropagation();
   254→    setPhotoDragOver(false);
   255→    const file = e.dataTransfer.files?.[0] || null;
   256→    handlePhotoSelect(file);
   257→  }, [handlePhotoSelect]);
   258→
   259→  const handlePhotoDragOver = useCallback((e: React.DragEvent) => {
   260→    e.preventDefault();
   261→    e.stopPropagation();
   262→    setPhotoDragOver(true);
   263→  }, []);
   264→
   265→  const handlePhotoDragLeave = useCallback((e: React.DragEvent) => {
   266→    e.preventDefault();
   267→    e.stopPropagation();
   268→    setPhotoDragOver(false);
   269→  }, []);
   270→
   271→  const updateField = <K extends keyof TvetFormData>(field: K, value: TvetFormData[K]) => {
   272→    setForm((prev) => ({ ...prev, [field]: value }));
   273→    if (errors[field]) setErrors((prev) => { const n = { ...prev }; delete n[field]; return n; });
   274→  };
   275→
   276→  const updateSubject = (section: 'pleSubjects' | 'olevelSubjects' | 'ujtcSubjects', idx: number, key: 'subject' | 'grade', value: string) => {
   277→    setForm((prev) => {
   278→      const arr = [...prev[section]];
   279→      arr[idx] = { ...arr[idx], [key]: value };
   280→      return { ...prev, [section]: arr };
   281→    });
   282→  };
   283→
   284→  const addSubjectRow = (section: 'pleSubjects' | 'olevelSubjects' | 'ujtcSubjects') => {
   285→    setForm((prev) => ({ ...prev, [section]: [...prev[section], { subject: '', grade: '' }] }));
   286→  };
   287→
   288→  const removeSubjectRow = (section: 'pleSubjects' | 'olevelSubjects' | 'ujtcSubjects', idx: number) => {
   289→    setForm((prev) => ({ ...prev, [section]: prev[section].filter((_, i) => i !== idx) }));
   290→  };
   291→
   292→  const updateOtherQual = (idx: number, key: keyof OtherQual, value: string) => {
   293→    setForm((prev) => {
   294→      const arr = [...prev.otherQuals];
   295→      arr[idx] = { ...arr[idx], [key]: value };
   296→      return { ...prev, otherQuals: arr };
   297→    });
   298→  };
   299→
   300→  const updateOtherQualSubject = (qualIdx: number, subIdx: number, key: 'subject' | 'grade', value: string) => {
   301→    setForm((prev) => {
   302→      const arr = [...prev.otherQuals];
   303→      const subs = [...arr[qualIdx].subjects];
   304→      subs[subIdx] = { ...subs[subIdx], [key]: value };
   305→      arr[qualIdx] = { ...arr[qualIdx], subjects: subs };
   306→      return { ...prev, otherQuals: arr };
   307→    });
   308→  };
   309→
   310→  const addOtherQual = () => {
   311→    setForm((prev) => ({
   312→      ...prev,
   313→      otherQuals: [...prev.otherQuals, {
   314→        institution: '', courseName: '', yearSitting: '', regNumber: '', classGrade: '',
   315→        subjects: [{ subject: '', grade: '' }, { subject: '', grade: '' }, { subject: '', grade: '' },
   316→                   { subject: '', grade: '' }, { subject: '', grade: '' }, { subject: '', grade: '' }],
   317→      }],
   318→    }));
   319→  };
   320→
   321→  const removeOtherQual = (idx: number) => {
   322→    setForm((prev) => ({ ...prev, otherQuals: prev.otherQuals.filter((_, i) => i !== idx) }));
   323→  };
   324→
   325→  const updateWorkRecord = (idx: number, key: keyof WorkRecord, value: string) => {
   326→    setForm((prev) => {
   327→      const arr = [...prev.workRecords];
   328→      arr[idx] = { ...arr[idx], [key]: value };
   329→      return { ...prev, workRecords: arr };
   330→    });
   331→  };
   332→
   333→  const addWorkRecord = () => {
   334→    setForm((prev) => ({ ...prev, workRecords: [...prev.workRecords, { organization: '', postHeld: '', period: '' }] }));
   335→  };
   336→
   337→  const removeWorkRecord = (idx: number) => {
   338→    setForm((prev) => ({ ...prev, workRecords: prev.workRecords.filter((_, i) => i !== idx) }));
   339→  };
   340→
   341→  const updateInstChoice = (idx: number, key: keyof InstitutionChoice, value: string) => {
   342→    setForm((prev) => {
   343→      const arr = [...prev.institutionChoices];
   344→      arr[idx] = { ...arr[idx], [key]: value };
   345→      return { ...prev, institutionChoices: arr };
   346→    });
   347→  };
   348→
   349→  /* ──────── validation ──────── */
   350→
   351→  const validateStep = (step: number): boolean => {
   352→    const e: Record<string, string> = {};
   353→    if (step === 1) {
   354→      if (!form.passportPhoto) e.passportPhoto = 'Passport photo is required';
   355→      if (!form.surname.trim()) e.surname = 'Surname is required';
   356→      if (!form.otherNames.trim()) e.otherNames = 'Other names are required';
   357→      if (!form.dob) e.dob = 'Date of birth is required';
   358→      if (!form.sex) e.sex = 'Sex is required';
   359→      if (!form.homeDistrict) e.homeDistrict = 'Home district is required';
   360→      if (!form.county) e.county = 'County is required';
   361→      if (!form.subCounty) e.subCounty = 'Sub-county is required';
   362→      if (!form.religion) e.religion = 'Religious affiliation is required';
   363→      if (!form.email.trim()) e.email = 'Email is required';
   364→      if (form.email && !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(form.email)) e.email = 'Invalid email address';
   365→      if (!form.telephone.trim()) e.telephone = 'Telephone contact is required';
   366→      if (!form.parentGuardianName.trim()) e.parentGuardianName = 'Parent/Guardian name is required';
   367→      if (!form.parentTelephone.trim()) e.parentTelephone = 'Parent telephone is required';
   368→    }
   369→    if (step === 2) {
   370→      if (!form.pleSchoolName.trim()) e.pleSchoolName = 'School name is required';
   371→      if (!form.pleYearSitting) e.pleYearSitting = 'Year is required';
   372→      if (!form.olevelSchoolName.trim()) e.olevelSchoolName = 'School name is required';
   373→      if (!form.olevelYearSitting) e.olevelYearSitting = 'Year is required';
   374→    }
   375→    if (step === 4) {
   376→      if (!form.institutionChoices[0].courseI.trim()) e['choice1courseI'] = 'First choice course is required';
   377→      if (!form.reasonForCourse.trim()) e.reasonForCourse = 'Reason for choosing course is required';
   378→    }
   379→    if (step === 5) {
   380→      if (!form.declarationName.trim()) e.declarationName = 'Declaration name is required';
   381→      if (!form.declarationDate) e.declarationDate = 'Date is required';
   382→    }
   383→    setErrors(e);
   384→    return Object.keys(e).length === 0;
   385→  };
   386→
   387→  const goNext = () => {
   388→    if (validateStep(currentStep)) {
   389→      if (currentStep < 5) setCurrentStep(currentStep + 1);
   390→    } else {
   391→      addToast('Please fill in all required fields', 'error');
   392→    }
   393→  };
   394→
   395→  const goPrev = () => {
   396→    if (currentStep > 1) setCurrentStep(currentStep - 1);
   397→  };
   398→
   399→  /* ──────── submit ──────── */
   400→
   401→  const handleSubmit = async () => {
   402→    if (!validateStep(5)) { addToast('Please fill in the declaration', 'error'); return; }
   403→    setSubmitting(true);
   404→    try {
   405→      // 1. Upload passport photo first
   406→      let passportPhotoUrl: string | undefined;
   407→      if (form.passportPhoto) {
   408→        const photoFormData = new FormData();
   409→        photoFormData.append('file', form.passportPhoto);
   410→        photoFormData.append('type', 'admission');
   411→        const uploadRes = await fetch('/api/upload', {
   412→          method: 'POST',
   413→          body: photoFormData,
   414→        });
   415→        if (uploadRes.ok) {
   416→          const uploadData = await uploadRes.json();
   417→          passportPhotoUrl = uploadData.url;
   418→        } else {
   419→          addToast('Failed to upload passport photo. Please try again.', 'error');
   420→          setSubmitting(false);
   421→          return;
   422→        }
   423→      }
   424→
   425→      // 2. Submit application with form data + photo URL
   426→      const programme = form.institutionChoices[0].courseI || 'TVET Programme';
   427→      const res = await fetch('/api/admissions/tvet', {
   428→        method: 'POST',
   429→        headers: { 'Content-Type': 'application/json' },
   430→        body: JSON.stringify({ ...form, passportPhotoUrl }),
   431→      });
   432→      const data = await res.json();
   433→      if (data.success) {
   434→        setResult(data.data);
   435→        setSubmitted(true);
   436→        addToast('TVET Application submitted successfully!', 'success');
   437→      } else {
   438→        addToast(data.message || 'Submission failed. Please try again.', 'error');
   439→      }
   440→    } catch {
   441→      addToast('Network error. Please check your connection and try again.', 'error');
   442→    } finally {
   443→      setSubmitting(false);
   444→    }
   445→  };
   446→
   447→  /* ──────── helpers ──────── */
   448→
   449→  const SectionTitle = ({ code, title }: { code: string; title: string }) => (
   450→    <div className="flex items-center gap-3 mb-6">
   451→      <Badge className="text-xs font-bold px-2.5 py-0.5" style={{ background: GOLD, color: PRIMARY }}>{code}</Badge>
   452→      <h3 className="text-lg font-bold text-gray-900">{title}</h3>
   453→    </div>
   454→  );
   455→
   456→  const FieldError = ({ field }: { field: string }) =>
   457→    errors[field] ? <p className="text-xs text-red-500 mt-1">{errors[field]}</p> : null;
   458→
   459→  const RequiredMark = () => <span className="text-red-500 ml-0.5">*</span>;
   460→
   461→  const SubjectGradeTable = ({
   462→    subjects, gradeOptions, section, maxRows = 20,
   463→  }: {
   464→    subjects: SubjectGrade[];
   465→    gradeOptions: string[];
   466→    section: 'pleSubjects' | 'olevelSubjects' | 'ujtcSubjects' | 'otherQualSubjects';
   467→    maxRows?: number;
   468→    qualIdx?: number;
   469→  }) => {
   470→    const handleAdd = () => {
   471→      if (section === 'otherQualSubjects' && qualIdx !== undefined) {
   472→        updateOtherQualSubject(qualIdx, subjects.length, 'subject', '');
   473→      } else {
   474→        addSubjectRow(section as 'pleSubjects' | 'olevelSubjects' | 'ujtcSubjects');
   475→      }
   476→    };
   477→    const handleRemove = (idx: number) => {
   478→      if (section === 'otherQualSubjects' && qualIdx !== undefined) {
   479→        setForm((prev) => {
   480→          const arr = [...prev.otherQuals];
   481→          arr[qualIdx] = { ...arr[qualIdx], subjects: arr[qualIdx].subjects.filter((_, i) => i !== idx) };
   482→          return { ...prev, otherQuals: arr };
   483→        });
   484→      } else {
   485→        removeSubjectRow(section as 'pleSubjects' | 'olevelSubjects' | 'ujtcSubjects', idx);
   486→      }
   487→    };
   488→    const handleUpdate = (idx: number, key: 'subject' | 'grade', value: string) => {
   489→      if (section === 'otherQualSubjects' && qualIdx !== undefined) {
   490→        updateOtherQualSubject(qualIdx, idx, key, value);
   491→      } else {
   492→        updateSubject(section as 'pleSubjects' | 'olevelSubjects' | 'ujtcSubjects', idx, key, value);
   493→      }
   494→    };
   495→
   496→    return (
   497→      <div className="mt-3">
   498→        <div className="rounded-lg border border-gray-200 overflow-hidden">
   499→          <Table>
   500→            <TableHeader>
   501→              <TableRow style={{ background: `${PRIMARY}10` }}>
   502→                <TableHead className="w-10 text-center text-xs font-bold" style={{ color: PRIMARY }}>#</TableHead>
   503→                <TableHead className="text-xs font-bold" style={{ color: PRIMARY }}>Subject</TableHead>
   504→                <TableHead className="w-40 text-xs font-bold" style={{ color: PRIMARY }}>Grade / Score</TableHead>
   505→                <TableHead className="w-10"></TableHead>
   506→              </TableRow>
