   250→/** Small per-field save indicator */
   251→function FieldIndicator({ state }: { state: 'idle' | 'saving' | 'saved' | 'error' }) {
   252→  return (
   253→    <AnimatePresence mode="wait">
   254→      {state !== 'idle' && (
   255→        <motion.span
   256→          key={state}
   257→          initial={{ opacity: 0, scale: 0.7, y: -4 }}
   258→          animate={{ opacity: 1, scale: 1, y: 0 }}
   259→          exit={{ opacity: 0, scale: 0.7, y: 4 }}
   260→          transition={{ duration: 0.2 }}
   261→          className="inline-flex items-center gap-1 text-xs font-medium shrink-0"
   262→        >
   263→          {state === 'saving' && (
   264→            <>
   265→              <Loader2 className="w-3 h-3 animate-spin text-[#1a3a6b]" />
   266→              <span className="text-[#1a3a6b]">Saving…</span>
   267→            </>
   268→          )}
   269→          {state === 'saved' && (
   270→            <>
   271→              <Check className="w-3 h-3 text-emerald-500" />
   272→              <span className="text-emerald-600">Saved</span>
   273→            </>
   274→          )}
   275→          {state === 'error' && (
   276→            <>
   277→              <AlertCircle className="w-3 h-3 text-red-500" />
   278→              <span className="text-red-500">Failed</span>
   279→            </>
   280→          )}
   281→        </motion.span>
   282→      )}
   283→    </AnimatePresence>
   284→  );
   285→}

   387→                      onChange={(v) => onValueChange(field.key, v)}
   388→                      onBlur={() => {
   389→                        if (field.type !== 'switch') onSaveField(field.key);
   390→                      }}
   391→                      onToggle={() => onSaveField(field.key)}
   392→                      spanFull={field.type === 'textarea' && (field.rows ?? 3) >= 5}
   393→                    />
   394→                  ))}
   395→                </div>

   402→  );
   403→}

   405→/** Individual field row */
   406→function FieldRow({
   407→  field,
   408→  value,
   409→  saveState,
   410→  onChange,
   411→  onBlur,
   412→  onToggle,
   413→  spanFull,
   414→}: {
   415→  field: FieldConfig;
   416→  value: string;
   417→  saveState: 'idle' | 'saving' | 'saved' | 'error';
   418→  onChange: (value: string) => void;
   419→  onBlur: () => void;
   420→  onToggle: () => void;
   421→  spanFull?: boolean;
   422→}) {
   423→  const isSwitch = field.type === 'switch';
   424→  const isChecked = value === 'true';
   425→
   426→  return (
   427→    <div className={`${spanFull ? 'md:col-span-2' : ''}`}>
   428→      <div className="flex items-start justify-between gap-3 mb-1.5">
   429→        <div className="flex items-center gap-2 min-w-0">
   430→          <Label
   431→            htmlFor={field.key}
   432→            className="text-sm font-medium text-slate-700 truncate"
   433→          >
   434→            {field.label}
   435→          </Label>
   436→          {field.maxLength && (
   437→            <span className="text-[10px] text-slate-300 font-mono shrink-0">
   438→              {value.length}/{field.maxLength}
   439→            </span>
   440→          )}
   441→        </div>
   442→        {!isSwitch && <FieldIndicator state={saveState} />}
   443→      </div>
   444→
   445→      {field.description && (
   446→        <p className="text-xs text-slate-400 mb-2 leading-relaxed">{field.description}</p>
   447→      )}
   448→
   449→      {isSwitch ? (
   450→        <div className="flex items-center justify-between p-3 rounded-lg bg-slate-50/80 border border-slate-100">
   451→          <span className="text-sm text-slate-600">{field.description || field.label}</span>
   452→          <div className="flex items-center gap-2">
   453→            <FieldIndicator state={saveState} />
   454→            <Switch
   455→              id={field.key}
   456→              checked={isChecked}
   457→              onCheckedChange={() => {
   458→                onChange(isChecked ? 'false' : 'true');
   459→                onToggle();
   460→              }}
   461→              className="data-[state=checked]:bg-[#1a3a6b]"
   462→            />
   463→          </div>
   464→        </div>
   465→      ) : field.type === 'textarea' ? (
   466→        <Textarea
   467→          id={field.key}
   468→          value={value}
   469→          onChange={(e) => onChange(e.target.value.slice(0, field.maxLength ?? 9999))}
   470→          onBlur={onBlur}
   471→          placeholder={field.placeholder}
   472→          rows={field.rows ?? 3}
   473→          className="resize-y min-h-[80px] text-sm border-slate-200 focus:border-[#1a3a6b] focus:ring-[#1a3a6b]/20"
   474→        />
   475→      ) : (
   476→        <Input
   477→          id={field.key}
   478→          type={field.type === 'url' || field.type === 'email' || field.type === 'phone' || field.type === 'number' ? 'text' : 'text'}
   479→          value={value}
   480→          onChange={(e) => onChange(e.target.value.slice(0, field.maxLength ?? 200))}
   481→          onBlur={onBlur}
   482→          placeholder={field.placeholder}
   483→          inputMode={field.type === 'number' ? 'numeric' : field.type === 'phone' ? 'tel' : field.type === 'url' ? 'url' : field.type === 'email' ? 'email' : undefined}
   484→          className="text-sm border-slate-200 focus:border-[#1a3a6b] focus:ring-[#1a3a6b]/20"
   485→        />
   486→      )}
   487→    </div>
   488→  );
   489→}

   497→  const [globalSaving, setGlobalSaving] = useState(false);
   498→  const [globalSaved, setGlobalSaved] = useState(false);
   499→  const [activeSection, setActiveSection] = useState<string | null>(null);
   500→  const [searchQuery, setSearchQuery] = useState('');
   501→  const debounceTimers = useRef<Record<string, ReturnType<typeof setTimeout>>>({});
   502→  const allKeysRef = useRef<string[]>(getAllKeys());
   503→
   504→  // ── Load all settings on mount ──
   505→  useEffect(() => {
   506→    const keys = allKeysRef.current;
   507→    fetch(`/api/settings/bulk?keys=${keys.join(',')}`)
   508→      .then((r) => r.json())
   509→      .then((data) => {
   510→        const fetched: Record<string, string> = data.settings || {};
   511→        // Merge with defaults
   512→        const merged: Record<string, string> = {};
   513→        for (const section of SETTINGS_SECTIONS) {
   514→          for (const field of section.fields) {
   515→            merged[field.key] =
   516→              fetched[field.key] !== null && fetched[field.key] !== undefined
   517→                ? fetched[field.key]
   518→                : field.defaultValue || '';
   519→          }
   520→        }
   521→        setValues(merged);
   522→        setLoading(false);
   523→      })
   524→      .catch(() => {
   525→        // Fallback to defaults
   526→        const defaults: Record<string, string> = {};
   527→        for (const section of SETTINGS_SECTIONS) {
   528→          for (const field of section.fields) {
   529→            defaults[field.key] = field.defaultValue || '';
   530→          }
   531→        }
   532→        setValues(defaults);
   533→        setLoading(false);
   534→      });
   535→  }, []);
   536→
   537→  // ── Cleanup debounce timers on unmount ──
   538→  useEffect(() => {
   539→    return () => {
   540→      Object.values(debounceTimers.current).forEach(clearTimeout);
   541→    };
   542→  }, []);
   543→
   544→  // ── Save a single setting (PUT /api/settings) ──
   545→  const saveField = useCallback(
   546→    async (key: string) => {
   547→      // Clear any existing debounce timer for this key
   548→      if (debounceTimers.current[key]) {
   549→        clearTimeout(debounceTimers.current[key]);
   550→      }
   551→
   552→      // Debounce: wait 800ms before saving (in case user is still typing)
   553→      debounceTimers.current[key] = setTimeout(async () => {
   554→        setSaveStates((prev) => ({ ...prev, [key]: 'saving' }));
   555→        try {
   556→          const res = await fetch('/api/settings', {
   557→            method: 'PUT',
   558→            headers: { 'Content-Type': 'application/json' },
   559→            body: JSON.stringify({ key, value: values[key] }),
   560→          });
   561→          if (res.ok) {
   562→            setSaveStates((prev) => ({ ...prev, [key]: 'saved' }));
   563→            // Reset to idle after 3 seconds
   564→            setTimeout(() => {
   565→              setSaveStates((prev) => ({ ...prev, [key]: 'idle' }));
   566→            }, 3000);
   567→          } else {
   568→            setSaveStates((prev) => ({ ...prev, [key]: 'error' }));
   569→            setTimeout(() => {
   570→              setSaveStates((prev) => ({ ...prev, [key]: 'idle' }));
   571→            }, 5000);
   572→          }
   573→        } catch {
   574→          setSaveStates((prev) => ({ ...prev, [key]: 'error' }));
   575→          setTimeout(() => {
   576→            setSaveStates((prev) => ({ ...prev, [key]: 'idle' }));
   577→          }, 5000);
   578→        }
   579→      }, 800);
   580→    },
   581→    [values]
   582→  );
   583→
   584→  // ── Save all settings at once (PUT /api/settings/bulk) ──
   585→  const saveAll = useCallback(async () => {
   586→    setGlobalSaving(true);
   587→    setGlobalSaved(false);
   588→    try {
   589→      const res = await fetch('/api/settings/bulk', {
   590→        method: 'PUT',
   591→        headers: { 'Content-Type': 'application/json' },
   592→        body: JSON.stringify({ settings: values }),
   593→      });
   594→      if (res.ok) {
   595→        setGlobalSaved(true);
   596→        // Mark all fields as saved
   597→        const allSaved: FieldSaveState = {};
   598→        for (const key of allKeysRef.current) {
   599→          allSaved[key] = 'saved';
   600→        }
   601→        setSaveStates(allSaved);
   602→        setTimeout(() => {
   603→          setGlobalSaved(false);
   604→          setSaveStates({});
   605→        }, 3000);
   606→      }
   607→    } catch {
   608→      // silently fail
   609→    } finally {
   610→      setGlobalSaving(false);
   611→    }
   612→  }, [values]);
   613→
   614→  // ── Handle value change ──
   615→  const handleValueChange = useCallback((key: string, value: string) => {
   616→    setValues((prev) => ({ ...prev, [key]: value }));
   617→    // Reset save state to idle when user starts editing
   618→    setSaveStates((prev) => ({ ...prev, [key]: 'idle' }));
   619→  }, []);
   620→
   621→  // ── Filter sections by search ──
   622→  const filteredSections = searchQuery.trim()
   623→    ? SETTINGS_SECTIONS.filter(
   624→        (section) =>
   625→          section.title.toLowerCase().includes(searchQuery.toLowerCase()) ||
   626→          section.fields.some(
   627→            (f) =>
   628→              f.label.toLowerCase().includes(searchQuery.toLowerCase()) ||
   629→              (values[f.key] ?? '').toLowerCase().includes(searchQuery.toLowerCase())
   630→          )
   631→      )
   632→    : SETTINGS_SECTIONS;
   633→
   634→  // ── Count filled settings ──
   635→  const filledCount = Object.values(values).filter((v) => v.trim() !== '').length;
   636→  const totalCount = allKeysRef.current.length;
   637→
   638→  return (
   639→    <div className="space-y-6">
   640→      {/* ── Header ── */}
   641→      <motion.div
   642→        initial={{ opacity: 0, y: -8 }}
   643→        animate={{ opacity: 1, y: 0 }}
   644→        transition={{ duration: 0.3 }}
   645→        className="flex flex-col sm:flex-row sm:items-center justify-between gap-4"
   646→      >
   647→        <div>
   648→          <h2 className="text-xl font-bold text-slate-800 flex items-center gap-2">
   649→            <div className="w-8 h-8 rounded-lg bg-[#1a3a6b]/10 flex items-center justify-center">
   650→              <Globe className="w-4.5 h-4.5 text-[#1a3a6b]" />
   651→            </div>
   652→            Website Content Management
   653→          </h2>
   654→          <p className="text-sm text-slate-400 mt-1">
   655→            Manage all website content, SEO, contact details, and feature toggles.
   656→          </p>
   657→        </div>
   658→        <div className="flex items-center gap-3">
   659→          <Badge
   660→            variant="outline"
   661→            className="text-xs font-normal text-slate-500 border-slate-200"
   662→          >
   663→            <Sparkles className="w-3 h-3 mr-1 text-[#f5c518]" />
   664→            {filledCount}/{totalCount} configured
   665→          </Badge>
   666→          <Button
   667→            onClick={saveAll}
   668→            disabled={globalSaving || loading}
   669→            style={{ backgroundColor: '#f5c518', color: '#1a3a6b' }}
   670→            className="shrink-0 disabled:opacity-50 hover:opacity-90 font-semibold shadow-sm hover:shadow-md transition-all"
   671→          >
   672→            {globalSaving ? (
   673→              <>
   674→                <Loader2 className="w-4 h-4 mr-2 animate-spin" />
   675→                Saving All…
   676→              </>
   677→            ) : globalSaved ? (
   678→              <>
   679→                <Check className="w-4 h-4 mr-2" />
   680→                All Saved!
   681→              </>
   682→            ) : (
   683→              <>
   684→                <Save className="w-4 h-4 mr-2" />
   685→                Save All
   686→              </>
   687→            )}
   688→          </Button>
   689→        </div>
   690→      </motion.div>
   691→
   692→      {/* ── Search & Section Nav ── */}
   693→      <motion.div
   694→        initial={{ opacity: 0, y: -4 }}
   695→        animate={{ opacity: 1, y: 0 }}
   696→        transition={{ duration: 0.3, delay: 0.1 }}
   697→        className="space-y-4"
   698→      >
   699→        {/* Search */}
   700→        <div className="relative">
   701→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-slate-400" />
   702→          <Input
   703→            value={searchQuery}
   704→            onChange={(e) => setSearchQuery(e.target.value)}
   705→            placeholder="Search settings…"
   706→            className="pl-9 text-sm border-slate-200 focus:border-[#1a3a6b] focus:ring-[#1a3a6b]/20"
   707→          />
   708→          {searchQuery && (
   709→            <button
   710→              onClick={() => setSearchQuery('')}
   711→              className="absolute right-3 top-1/2 -translate-y-1/2 text-slate-400 hover:text-slate-600"
   712→            >
   713→              <span className="text-xs">Clear</span>
   714→            </button>
   715→          )}
   716→        </div>
   717→
   718→        {/* Section quick-nav chips */}
   719→        <div className="flex flex-wrap gap-2">
   720→          {SETTINGS_SECTIONS.map((section) => {
   721→            const Icon = section.icon;
   722→            const isActive = activeSection === section.id;
   723→            return (
   724→              <button
   725→                key={section.id}
   726→                onClick={() =>
   727→                  setActiveSection(isActive ? null : section.id)
   728→                }
   729→                className={`inline-flex items-center gap-1.5 px-3 py-1.5 rounded-full text-xs font-medium transition-all duration-200 border ${
   730→                  isActive
   731→                    ? 'bg-[#1a3a6b] text-white border-[#1a3a6b] shadow-sm'
   732→                    : 'bg-white text-slate-600 border-slate-200 hover:border-[#1a3a6b] hover:text-[#1a3a6b]'
   733→                }`}
   734→              >
   735→                <Icon className="w-3 h-3" />
   736→                {section.title}
   737→              </button>
   738→            );
   739→          })}
   740→        </div>
   741→      </motion.div>
   742→
   743→      {/* ── Auto-save notice ── */}
   744→      <motion.div
   745→        initial={{ opacity: 0 }}
   746→        animate={{ opacity: 1 }}
   747→        transition={{ delay: 0.3 }}
   748→        className="flex items-center gap-2 px-4 py-2.5 rounded-lg bg-[#f5c518]/10 border border-[#f5c518]/20"
   749→      >
   750→        <Info className="w-4 h-4 text-[#f5c518] shrink-0" />
   751→        <p className="text-xs text-amber-800/80">
   752→          Text fields auto-save on blur (with debounce). Switches save immediately. You can also use{' '}
   753→          <span className="font-semibold">Save All</span> to persist everything at once.
   754→        </p>
   755→      </motion.div>
   756→
   757→      {/* ── Loading State ── */}
   758→      {loading && (
   759→        <div className="space-y-4">
   760→          <SectionSkeleton />
   761→          <SectionSkeleton />
   762→          <SectionSkeleton />
   763→        </div>
   764→      )}
   765→
   766→      {/* ── Settings Sections ── */}
   767→      {!loading && (
   768→        <div className="space-y-4">
   769→          <AnimatePresence>
   770→            {filteredSections
   771→              .filter((section) => !activeSection || section.id === activeSection)
   772→              .map((section) => (
   773→                <SettingsSectionCard
   774→                  key={section.id}
   775→                  section={section}
   776→                  values={values}
   777→                  saveStates={saveStates}
   778→                  onValueChange={handleValueChange}
   779→                  onSaveField={saveField}
   780→                />
   781→              ))}
   782→          </AnimatePresence>
   783→
   784→          {filteredSections.filter(
   785→            (section) => !activeSection || section.id === activeSection
   786→          ).length === 0 && (
   787→            <motion.div
   788→              initial={{ opacity: 0 }}
   789→              animate={{ opacity: 1 }}
   790→              className="text-center py-12"
   791→            >
   792→              <Search className="w-10 h-10 text-slate-300 mx-auto mb-3" />
   793→              <p className="text-sm text-slate-400">No settings match &quot;{searchQuery}&quot;</p>
   794→              <Button
   795→                variant="link"
   796→                size="sm"
   797→                className="text-[#1a3a6b] mt-2"
   798→                onClick={() => setSearchQuery('')}
   799→              >
   800→                Clear search
   801→              </Button>
   802→            </motion.div>
   803→          )}
   804→        </div>
   805→      )}
   806→
   807→      {/* ── Bottom Action Bar ── */}
   808→      {!loading && (
   809→        <motion.div
   810→          initial={{ opacity: 0, y: 8 }}
   811→          animate={{ opacity: 1, y: 0 }}
   812→          transition={{ delay: 0.2 }}
   813→          className="sticky bottom-4 z-10"
   814→        >
   815→          <div className="flex items-center justify-between p-4 rounded-xl bg-white/90 backdrop-blur-sm border border-slate-200 shadow-lg">
   816→            <div className="flex items-center gap-3">
   817→              <div className="w-2 h-2 rounded-full bg-emerald-400 animate-pulse" />
   818→              <span className="text-xs text-slate-500">
   819→                {filledCount} of {totalCount} fields configured
   820→              </span>
   821→              <Separator orientation="vertical" className="h-4" />
   822→              <span className="text-xs text-slate-400">
   823→                Last auto-save: {new Date().toLocaleTimeString()}
   824→              </span>
   825→            </div>
   826→            <div className="flex items-center gap-2">
   827→              <Button
   828→                variant="outline"
   829→                size="sm"
   830→                className="text-xs border-slate-200 text-slate-600 hover:border-[#1a3a6b] hover:text-[#1a3a6b]"
   831→                onClick={() => {
   832→                  // Reset to defaults
   833→                  const defaults: Record<string, string> = {};
   834→                  for (const section of SETTINGS_SECTIONS) {
   835→                    for (const field of section.fields) {
   836→                      defaults[field.key] = field.defaultValue || '';
   837→                    }
   838→                  }
   839→                  setValues(defaults);
   840→                }}
   841→              >
   842→                <RefreshCw className="w-3 h-3 mr-1" />
   843→                Reset to Defaults
   844→              </Button>
   845→              <Button
   846→                size="sm"
   847→                onClick={saveAll}
   848→                disabled={globalSaving}
   849→                className="bg-[#1a3a6b] hover:bg-[#2756a0] text-white text-xs font-semibold shadow-sm disabled:opacity-50"
   850→              >
   851→                {globalSaving ? (
   852→                  <>
   853→                    <Loader2 className="w-3 h-3 mr-1 animate-spin" />
   854→                    Saving…
   855→                  </>
   856→                ) : (
   857→                  <>
   858→                    <Save className="w-3 h-3 mr-1" />
   859→                    Save All Changes
   860→                  </>
   861→                )}
   862→              </Button>
   863→            </div>
   864→          </div>
   865→        </motion.div>
   866→      )}
   867→    </div>
   868→  );
   869→}
