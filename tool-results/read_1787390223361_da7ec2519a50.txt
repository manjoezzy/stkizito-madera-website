    96→/* ══════════════════════════════════════════════════════
    97→   ANIMATED COUNTER HOOK
    98→   ══════════════════════════════════════════════════════ */
    99→function useCountUp(target: number, duration = 2000, start = false) {
   100→  const [count, setCount] = useState(0);
   101→  const frameRef = useRef<number>(0);
   102→
   103→  useEffect(() => {
   104→    if (!start) return;
   105→    let startTime: number | null = null;
   106→
   107→    const animate = (timestamp: number) => {
   108→      if (!startTime) startTime = timestamp;
   109→      const progress = Math.min((timestamp - startTime) / duration, 1);
   110→      const eased = 1 - Math.pow(1 - progress, 3); // easeOutCubic
   111→      setCount(Math.floor(eased * target));
   112→      if (progress < 1) {
   113→        frameRef.current = requestAnimationFrame(animate);
   114→      }
   115→    };
   116→
   117→    frameRef.current = requestAnimationFrame(animate);
   118→    return () => cancelAnimationFrame(frameRef.current);
   119→  }, [target, duration, start]);
   120→
   121→  return count;
   122→}
