   101→/* ══════════════════════════════════════════════════════
   102→   ANIMATED COUNTER HOOK
   103→   ══════════════════════════════════════════════════════ */
   104→function useCountUp(target: number, duration = 2000, start = false) {
   105→  const [count, setCount] = useState(0);
   106→  const frameRef = useRef<number>(0);
   107→
   108→  useEffect(() => {
   109→    if (!start) return;
   110→    let startTime: number | null = null;
   111→
   112→    const animate = (timestamp: number) => {
   113→      if (!startTime) startTime = timestamp;
   114→      const progress = Math.min((timestamp - startTime) / duration, 1);
   115→      const eased = 1 - Math.pow(1 - progress, 3); // easeOutCubic
   116→      setCount(Math.floor(eased * target));
   117→      if (progress < 1) {
   118→        frameRef.current = requestAnimationFrame(animate);
   119→      }
   120→    };
   121→
   122→    frameRef.current = requestAnimationFrame(animate);
   123→    return () => cancelAnimationFrame(frameRef.current);
   124→  }, [target, duration, start]);
   125→
   126→  return count;
   127→}
