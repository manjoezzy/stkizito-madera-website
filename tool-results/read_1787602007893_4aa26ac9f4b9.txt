    52→// ===================== TYPES =====================

    60→  thumbnailUrl: string | null;
    61→  ceremonyYear: string;

    84→  title: string;
    85→  description: string;
    86→  itemType: 'photo' | 'video';
    87→  mediaUrl: string;
    88→  thumbnailUrl: string;
    89→  ceremonyYear: string;
    90→  ceremonyName: string;
    91→  sortOrder: number;
    92→}

   133→// ===================== COMPONENT =====================
   134→
   135→export default function AdminGraduationSection() {
   136→  // ---- Data State ----
   137→  const [items, setItems] = useState<GraduationItem[]>([]);
   138→  const [total, setTotal] = useState(0);
   139→  const [totalPages, setTotalPages] = useState(0);
   140→  const [loading, setLoading] = useState(true);
   141→
   142→  // ---- Filter State ----
   143→  const [searchQuery, setSearchQuery] = useState('');
   144→  const [filterYear, setFilterYear] = useState<string>('all');
   145→  const [filterType, setFilterType] = useState<string>('all');
   146→  const [availableYears, setAvailableYears] = useState<string[]>([]);
   147→
   148→  // ---- Pagination State ----
   149→  const [currentPage, setCurrentPage] = useState(1);
   150→
   151→  // ---- Dialog State ----
   152→  const [dialogOpen, setDialogOpen] = useState(false);
   153→  const [editingItem, setEditingItem] = useState<GraduationItem | null>(null);
   154→  const [formData, setFormData] = useState<FormData>({ ...EMPTY_FORM });
   155→
   156→  // ---- Upload State ----
   157→  const [uploading, setUploading] = useState(false);
   158→  const [uploadPreview, setUploadPreview] = useState<string | null>(null);
   159→  const [uploadFileName, setUploadFileName] = useState<string>('');
   160→  const fileInputRef = useRef<HTMLInputElement>(null);
   161→
   162→  // ---- Action State ----
   163→  const [saving, setSaving] = useState(false);
   164→  const [deleting, setDeleting] = useState<string | null>(null);
   165→  const [togglingPublish, setTogglingPublish] = useState<string | null>(null);
   166→
   167→  // ===================== DATA FETCHING =====================
   168→
   169→  const fetchItems = useCallback(async () => {
   170→    setLoading(true);
   171→    try {
   172→      const params = new URLSearchParams();
   173→      params.set('page', currentPage.toString());
   174→      params.set('limit', ITEMS_PER_PAGE.toString());
   175→      if (filterType !== 'all') params.set('type', filterType);
   176→      if (filterYear !== 'all') params.set('year', filterYear);
   177→
   178→      const res = await fetch(`/api/graduation?${params.toString()}`);
   179→      if (!res.ok) throw new Error('Failed to fetch graduation items');
   180→
   181→      const json: GraduationApiResponse = await res.json();
   182→      let filtered = json.data;
   183→
   184→      // Client-side search filter
   185→      if (searchQuery.trim()) {
   186→        const q = searchQuery.toLowerCase();
   187→        filtered = filtered.filter(
   188→          (item) =>
   189→            item.title.toLowerCase().includes(q) ||
   190→            (item.description && item.description.toLowerCase().includes(q)) ||
   191→            (item.ceremonyName && item.ceremonyName.toLowerCase().includes(q))
   192→        );
   193→      }
   194→
   195→      setItems(filtered);
   196→      setTotal(json.total);
   197→      setTotalPages(json.totalPages);
   198→
   199→      // Extract unique years from data
   200→      const years = new Set<string>();
   201→      json.data.forEach((item) => years.add(item.ceremonyYear));
   202→      setAvailableYears(
   203→        Array.from(years).sort((a, b) => b.localeCompare(a))
   204→      );
   205→    } catch (err) {
   206→      console.error('Error fetching graduation items:', err);
   207→    } finally {
   208→      setLoading(false);
   209→    }
   210→  }, [currentPage, filterType, filterYear, searchQuery]);
   211→
   212→  useEffect(() => {
   213→    fetchItems();
   214→  }, [fetchItems]);
   215→
   216→  // ===================== COMPUTED STATS =====================
   217→
   218→  const stats = React.useMemo(() => {
   219→    const photoCount = items.filter((i) => i.itemType === 'photo').length;
   220→    const videoCount = items.filter((i) => i.itemType === 'video').length;
   221→    const publishedCount = items.filter((i) => i.isPublished).length;
   222→    return { total: items.length, photoCount, videoCount, publishedCount };
   223→  }, [items]);
   224→
   225→  // ===================== HANDLERS =====================
   226→
   227→  const openAddDialog = () => {
   228→    setEditingItem(null);
   229→    setFormData({ ...EMPTY_FORM, ceremonyYear: new Date().getFullYear().toString() });
   230→    setUploadPreview(null);
   231→    setUploadFileName('');
   232→    setDialogOpen(true);
   233→  };
   234→
   235→  const openEditDialog = (item: GraduationItem) => {
   236→    setEditingItem(item);
   237→    setFormData({
   238→      title: item.title,
   239→      description: item.description || '',
   240→      itemType: item.itemType as 'photo' | 'video',
   241→      mediaUrl: item.mediaUrl,
   242→      thumbnailUrl: item.thumbnailUrl || '',
   243→      ceremonyYear: item.ceremonyYear,
   244→      ceremonyName: item.ceremonyName || '',
   245→      sortOrder: item.sortOrder,
   246→    });
   247→    setUploadPreview(item.itemType === 'photo' ? item.mediaUrl : (item.thumbnailUrl || null));
   248→    setUploadFileName('');
   249→    setDialogOpen(true);
   250→  };
   251→
   252→  const handleFileUpload = async (file: File) => {
   253→    // Validate file
   254→    const isImage = file.type.startsWith('image/');
   255→    const isVideo = ['video/mp4', 'video/webm', 'video/ogg'].includes(file.type);
   256→
   257→    if (!isImage && !isVideo) {
   258→      alert('Unsupported file type. Please upload an image (JPG, PNG, WebP, GIF) or video (MP4, WebM, OGG).');
   259→      return;
   260→    }
   261→
   262→    if (file.size > 10 * 1024 * 1024) {
   263→      alert('File size exceeds 10MB limit.');
   264→      return;
   265→    }
   266→
   267→    setUploading(true);
   268→    try {
   269→      const fd = new FormData();
   270→      fd.append('file', file);
   271→      fd.append('type', 'graduation');
   272→
   273→      const res = await fetch('/api/upload', { method: 'POST', body: fd });
   274→      if (!res.ok) throw new Error('Upload failed');
   275→
   276→      const data: UploadResponse = await res.json();
   277→      setFormData((prev) => ({
   278→        ...prev,
   279→        mediaUrl: data.url,
   280→        itemType: isVideo ? 'video' : prev.itemType,
   281→      }));
   282→      setUploadFileName(data.fileName);
   283→
   284→      // Generate preview
   285→      if (isImage) {
   286→        setUploadPreview(data.url);
   287→      } else {
   288→        const videoEl = document.createElement('video');
   289→        videoEl.src = data.url;
   290→        videoEl.currentTime = 1;
   291→        videoEl.addEventListener('seeked', () => {
   292→          const canvas = document.createElement('canvas');
   293→          canvas.width = videoEl.videoWidth || 640;
   294→          canvas.height = videoEl.videoHeight || 360;
   295→          const ctx = canvas.getContext('2d');
   296→          if (ctx) {
   297→            ctx.drawImage(videoEl, 0, 0, canvas.width, canvas.height);
   298→            setUploadPreview(canvas.toDataURL('image/jpeg', 0.7));
   299→          }
   300→        }, { once: true });
   301→      }
   302→    } catch (err) {
   303→      console.error('Upload error:', err);
   304→      alert('Failed to upload file. Please try again.');
   305→    } finally {
   306→      setUploading(false);
   307→    }
   308→  };
   309→
   310→  const handleSave = async () => {
   311→    if (!formData.title.trim()) {
   312→      alert('Title is required.');
   313→      return;
   314→    }
   315→    if (!formData.mediaUrl.trim()) {
   316→      alert('Please upload a media file.');
   317→      return;
   318→    }
   319→
   320→    setSaving(true);
   321→    try {
   322→      if (editingItem) {
   323→        // Update
   324→        const res = await fetch('/api/graduation', {
   325→          method: 'PUT',
   326→          headers: { 'Content-Type': 'application/json' },
   327→          body: JSON.stringify({ id: editingItem.id, ...formData }),
   328→        });
   329→        if (!res.ok) throw new Error('Update failed');
   330→      } else {
   331→        // Create
   332→        const res = await fetch('/api/graduation', {
   333→          method: 'POST',
   334→          headers: { 'Content-Type': 'application/json' },
   335→          body: JSON.stringify(formData),
   336→        });
   337→        if (!res.ok) throw new Error('Create failed');
   338→      }
   339→
   340→      setDialogOpen(false);
   341→      fetchItems();
   342→    } catch (err) {
   343→      console.error('Save error:', err);
   344→      alert('Failed to save. Please try again.');
   345→    } finally {
   346→      setSaving(false);
   347→    }
   348→  };
   349→
   350→  const handleDelete = async (item: GraduationItem) => {
   351→    if (!window.confirm(`Are you sure you want to delete "${item.title}"? This action cannot be undone.`)) {
   352→      return;
   353→    }
   354→
   355→    setDeleting(item.id);
   356→    try {
   357→      const res = await fetch('/api/graduation', {
   358→        method: 'DELETE',
   359→        headers: { 'Content-Type': 'application/json' },
   360→        body: JSON.stringify({ id: item.id }),
   361→      });
   362→      if (!res.ok) throw new Error('Delete failed');
   363→      fetchItems();
   364→    } catch (err) {
   365→      console.error('Delete error:', err);
   366→      alert('Failed to delete. Please try again.');
   367→    } finally {
   368→      setDeleting(null);
   369→    }
   370→  };
   371→
   372→  const handleTogglePublish = async (item: GraduationItem) => {
   373→    setTogglingPublish(item.id);
   374→    try {
   375→      const res = await fetch('/api/graduation', {
   376→        method: 'PUT',
   377→        headers: { 'Content-Type': 'application/json' },
   378→        body: JSON.stringify({ id: item.id, isPublished: !item.isPublished }),
   379→      });
   380→      if (!res.ok) throw new Error('Toggle failed');
   381→      fetchItems();
   382→    } catch (err) {
   383→      console.error('Toggle publish error:', err);
   384→      alert('Failed to update publish status.');
   385→    } finally {
   386→      setTogglingPublish(null);
   387→    }
   388→  };
   389→
   390→  const handleDrop = (e: React.DragEvent) => {
   391→    e.preventDefault();
   392→    const file = e.dataTransfer.files[0];
   393→    if (file) handleFileUpload(file);
   394→  };
   395→
   396→  const handleDragOver = (e: React.DragEvent) => {
   397→    e.preventDefault();
   398→  };
   399→
   400→  const resetFilters = () => {
   401→    setSearchQuery('');
   402→    setFilterYear('all');
   403→    setFilterType('all');
   404→    setCurrentPage(1);
   405→  };
   406→
   407→  // ===================== RENDER HELPERS =====================
   408→
   409→  const renderMediaCard = (item: GraduationItem) => {
   410→    const isVideo = item.itemType === 'video';
   411→    const imageSrc = isVideo ? item.thumbnailUrl : item.mediaUrl;
   412→
   413→    return (
   414→      <motion.div key={item.id} variants={cardVariants} layout>
   415→        <Card className="border-0 shadow-sm overflow-hidden group hover:shadow-md transition-shadow duration-200">
   416→          {/* Media Preview */}
   417→          <div className="relative aspect-video bg-muted/50 overflow-hidden">
   418→            {imageSrc ? (
   419→              <img
   420→                src={imageSrc}
   421→                alt={item.title}
   422→                className="w-full h-full object-cover transition-transform duration-300 group-hover:scale-105"
   423→              />
   424→            ) : (
   425→              <div className="w-full h-full flex items-center justify-center bg-muted">
   426→                {isVideo ? (
   427→                  <Film className="w-12 h-12 text-muted-foreground/40" />
   428→                ) : (
   429→                  <ImageIcon className="w-12 h-12 text-muted-foreground/40" />
   430→                )}
   431→              </div>
   432→            )}
   433→
   434→            {/* Play icon overlay for videos */}
   435→            {isVideo && (
   436→              <div className="absolute inset-0 flex items-center justify-center">
   437→                <div
   438→                  className="w-14 h-14 rounded-full flex items-center justify-center bg-black/50 backdrop-blur-sm group-hover:bg-black/70 transition-colors"
   439→                >
   440→                  <Play className="w-7 h-7 text-white fill-white ml-1" />
   441→                </div>
   442→              </div>
   443→            )}
   444→
   445→            {/* Type badge */}
   446→            <div className="absolute top-2 left-2">
   447→              <Badge
   448→                variant="secondary"
   449→                className="text-[10px] font-semibold uppercase tracking-wider backdrop-blur-md bg-black/50 text-white border-0"
   450→              >
   451→                {isVideo ? 'Video' : 'Photo'}
   452→              </Badge>
   453→            </div>
   454→
   455→            {/* Published indicator */}
   456→            <div className="absolute top-2 right-2">
   457→              {item.isPublished ? (
   458→                <Badge
   459→                  className="text-[10px] font-semibold bg-green-600 text-white border-0"
   460→                >
   461→                  <CheckCircle2 className="w-3 h-3 mr-1" />
   462→                  Live
   463→                </Badge>
   464→              ) : (
   465→                <Badge
   466→                  variant="secondary"
   467→                  className="text-[10px] font-semibold backdrop-blur-md bg-black/50 text-white border-0"
   468→                >
   469→                  <XCircle className="w-3 h-3 mr-1" />
   470→                  Draft
   471→                </Badge>
   472→              )}
   473→            </div>
   474→
   475→            {/* Hover actions overlay */}
   476→            <div className="absolute inset-0 bg-gradient-to-t from-black/60 via-transparent to-transparent opacity-0 group-hover:opacity-100 transition-opacity duration-200">
   477→              <div className="absolute bottom-2 right-2 flex gap-1.5">
   478→                <button
   479→                  onClick={() => openEditDialog(item)}
   480→                  className="p-1.5 rounded-md bg-white/90 hover:bg-white text-gray-800 transition-colors"
   481→                  title="Edit"
   482→                >
   483→                  <Pencil className="w-3.5 h-3.5" />
   484→                </button>
   485→                <button
   486→                  onClick={() => handleDelete(item)}
   487→                  disabled={deleting === item.id}
   488→                  className="p-1.5 rounded-md bg-red-500/90 hover:bg-red-600 text-white transition-colors disabled:opacity-50"
   489→                  title="Delete"
   490→                >
   491→                  {deleting === item.id ? (
   492→                    <Loader2 className="w-3.5 h-3.5 animate-spin" />
   493→                  ) : (
   494→                    <Trash2 className="w-3.5 h-3.5" />
   495→                  )}
   496→                </button>
   497→              </div>
   498→            </div>
   499→          </div>
   500→
   501→          {/* Card Body */}
   502→          <CardContent className="p-3 space-y-2">
   503→            <div className="flex items-start justify-between gap-2">
   504→              <h3 className="text-sm font-semibold text-gray-900 leading-tight line-clamp-2">
   505→                {item.title}
   506→              </h3>
   507→              <Badge
   508→                className="shrink-0 text-[10px] font-bold px-1.5 py-0"
   509→                style={{
   510→                  backgroundColor: PRIMARY_BLUE,
   511→                  color: 'white',
   512→                }}
   513→              >
   514→                {item.ceremonyYear}
   515→              </Badge>
   516→            </div>
   517→
   518→            {item.ceremonyName && (
   519→              <p className="text-xs text-muted-foreground truncate">
   520→                {item.ceremonyName}
   521→              </p>
   522→            )}
   523→
   524→            {item.description && (
   525→              <p className="text-xs text-muted-foreground line-clamp-2">
   526→                {item.description}
   527→              </p>
   528→            )}
   529→
   530→            {/* Actions row */}
   531→            <div className="flex items-center justify-between pt-1 border-t border-border/50">
   532→              <div className="flex items-center gap-1.5 text-[11px] text-muted-foreground">
   533→                {isVideo ? (
   534→                  <FileVideo className="w-3 h-3" />
   535→                ) : (
   536→                  <Camera className="w-3 h-3" />
   537→                )}
   538→                <span className="capitalize">{item.itemType}</span>
   539→                {item.sortOrder > 0 && (
   540→                  <span className="text-muted-foreground/60">· Order: {item.sortOrder}</span>
   541→                )}
   542→              </div>
   543→
   544→              <button
   545→                onClick={() => handleTogglePublish(item)}
   546→                disabled={togglingPublish === item.id}
   547→                className="flex items-center gap-1 text-[11px] font-medium transition-colors disabled:opacity-50"
   548→                style={{
   549→                  color: item.isPublished ? '#16a34a' : '#9ca3af',
   550→                }}
   551→              >
   552→                {togglingPublish === item.id ? (
   553→                  <Loader2 className="w-3.5 h-3.5 animate-spin" />
   554→                ) : item.isPublished ? (
   555→                  <Eye className="w-3.5 h-3.5" />
   556→                ) : (
   557→                  <EyeOff className="w-3.5 h-3.5" />
   558→                )}
   559→                {item.isPublished ? 'Published' : 'Draft'}
   560→              </button>
   561→            </div>
   562→          </CardContent>
   563→        </Card>
   564→      </motion.div>
   565→    );
   566→  };
   567→
   568→  // ===================== MAIN RENDER =====================
   569→
   570→  return (
   571→    <section className="space-y-6">
   572→      {/* ===== HEADER ===== */}
   573→      <motion.div
   574→        initial={{ opacity: 0, y: -10 }}
   575→        animate={{ opacity: 1, y: 0 }}
   576→        transition={{ duration: 0.3 }}
   577→        className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4"
   578→      >
   579→        <div>
   580→          <div className="flex items-center gap-2.5 mb-1">
   581→            <div
   582→              className="w-9 h-9 rounded-lg flex items-center justify-center"
   583→              style={{ backgroundColor: `${PRIMARY_BLUE}15` }}
   584→            >
   585→              <GraduationCap className="w-5 h-5" style={{ color: PRIMARY_BLUE }} />
   586→            </div>
   587→            <h2 className="text-2xl font-bold tracking-tight" style={{ color: PRIMARY_BLUE }}>
   588→              Graduation Gallery
   589→            </h2>
   590→          </div>
   591→          <p className="text-sm text-muted-foreground ml-[46px]">
   592→            Manage graduation photos and videos displayed on the public graduation page
   593→          </p>
   594→        </div>
   595→        <Button
   596→          onClick={openAddDialog}
   597→          className="shrink-0 text-white font-semibold shadow-sm hover:shadow-md transition-shadow"
   598→          style={{ backgroundColor: PRIMARY_BLUE }}
   599→          onMouseEnter={(e) => (e.currentTarget.style.backgroundColor = '#244a8a')}
   600→          onMouseLeave={(e) => (e.currentTarget.style.backgroundColor = PRIMARY_BLUE)}
   601→        >
   602→          <Plus className="w-4 h-4 mr-2" />
