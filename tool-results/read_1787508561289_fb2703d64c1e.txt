   162→/* ──────────────── component ──────────────── */
   163→
   164→export default function AdmissionsPage() {
   165→  const { addToast, setCurrentPage } = useAppStore();
   166→
   167→  const [nonFormalFormUrl, setNonFormalFormUrl] = useState<string | null>(null);
   168→  const [tvetFormUrl, setTvetFormUrl] = useState<string>('/forms/tvet-admission-form.pdf');
   169→  const [applicationFee, setApplicationFee] = useState<number>(50000); // default
   170→  useEffect(() => {
   171→    Promise.all([
   172→      fetch('/api/settings?key=non_formal_form_url').then((r) => r.json()),
   173→      fetch('/api/settings?key=tvet_form_url').then((r) => r.json()),
   174→      fetch('/api/settings?key=application_fee').then((r) => r.json()),
   175→    ]).then(([nf, tf, af]) => {
   176→      if (nf.value) setNonFormalFormUrl(nf.value);
   177→      if (tf.value) setTvetFormUrl(tf.value);
   178→      if (af.value) setApplicationFee(parseInt(af.value, 10) || 50000);
   179→    }).catch(() => {});
   180→  }, []);
   181→
   182→  const [currentStep, setCurrentStep] = useState(1);
   183→  const [form, setForm] = useState<FormData>(INITIAL_FORM);
   184→  const [errors, setErrors] = useState<Partial<Record<keyof FormData, string>>>({});
   185→  const [submitting, setSubmitting] = useState(false);
   186→  const [referenceNumber, setReferenceNumber] = useState('');
   187→  const [paymentStatus, setPaymentStatus] = useState<'idle' | 'initiating' | 'dialog' | 'verifying' | 'success' | 'failed'>('idle');
   188→  const [transactionRef, setTransactionRef] = useState('');
   189→  const [paymentError, setPaymentError] = useState('');
   190→  const [schoolPayCode, setSchoolPayCode] = useState('');
   191→  const [uploadedDocs, setUploadedDocs] = useState<{ name: string; url: string; size: number }[]>([]);
   192→  const [uploadingDoc, setUploadingDoc] = useState(false);
   193→
   194→  const updateField = (field: keyof FormData, value: string) => {
   195→    setForm((prev) => ({ ...prev, [field]: value }));
   196→    if (errors[field]) {
   197→      setErrors((prev) => {
   198→        const next = { ...prev };
   199→        delete next[field];
   200→        return next;
   201→      });
   202→    }
   203→  };
   204→
   205→
   206→  /* ──────── validation ──────── */
   207→
   208→  const validateStep = (step: number): boolean => {
   209→    const e: Partial<Record<keyof FormData, string>> = {};
   210→
   211→    if (step === 1) {
   212→      if (!form.fullName.trim()) e.fullName = 'Full name is required';
   213→      if (!form.dob) e.dob = 'Date of birth is required';
   214→      if (!form.gender) e.gender = 'Gender is required';
   215→      if (!form.nationality.trim()) e.nationality = 'Nationality is required';
   216→    }
   217→
   218→    if (step === 2) {
   219→      if (!form.phone.trim()) e.phone = 'Phone number is required';
   220→      else if (!/^\d{10}$/.test(form.phone.replace(/\s/g, ''))) e.phone = 'Enter a valid 10-digit phone number';
   221→      if (!form.email.trim()) e.email = 'Email is required';
   222→      else if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(form.email)) e.email = 'Enter a valid email address';
   223→    }
   224→
   225→    if (step === 3) {
   226→      if (!form.qualification) e.qualification = 'Select your highest qualification';
   227→    }
   228→
   229→    if (step === 4) {
   230→      if (!form.programme) e.programme = 'Select a programme';
   231→      if (!form.intakeYear) e.intakeYear = 'Select intake year';
   232→    }
   233→
   234→    if (step === 5) {
   235→      // No validation - documents optional but show warning
   236→    }
   237→
   238→    setErrors(e);
   239→    return Object.keys(e).length === 0;
   240→  };
   241→
   242→  const goNext = () => {
   243→    if (!validateStep(currentStep)) return;
   244→    if (currentStep === 4) {
   245→      // After programme selection, move to document upload
   246→      setCurrentStep((s) => Math.min(s + 1, 6));
   247→    } else if (currentStep === 5) {
   248→      // After documents, submit application and go to payment
   249→      handleSubmitApplication();
   250→    } else {
   251→      setCurrentStep((s) => Math.min(s + 1, 6));
   252→    }
   253→  };
   254→
   255→  const goPrev = () => {
   256→    setCurrentStep((s) => Math.max(s - 1, 1));
   257→  };
   258→
   259→  /* ──────── submit application (step 4 → step 5) ──────── */
   260→
   261→  const handleSubmitApplication = async () => {
   262→    setSubmitting(true);
   263→    try {
   264→      const res = await fetch('/api/admissions', {
   265→        method: 'POST',
   266→        headers: { 'Content-Type': 'application/json' },
   267→        body: JSON.stringify(form),
   268→      });
   269→      const data = await res.json();
   270→
   271→      if (data.success) {
   272→        setReferenceNumber(data.data.referenceNumber);
   273→        // Generate SchoolPay code from applicant summary info
   274→        const nameParts = form.fullName.trim().split(/\s+/);
   275→        const initials = nameParts.map((n: string) => n[0]).join('').toUpperCase().slice(0, 3);
   276→        const yearSuffix = new Date().getFullYear().toString().slice(-2);
   277→        const rand = Math.floor(1000 + Math.random() * 9000);
   278→        const code = `SKTM${yearSuffix}${initials}${rand}`;
   279→        setSchoolPayCode(code);
   280→        setCurrentStep(6);
   281→        addToast('Application submitted! Your SchoolPay code has been generated.', 'success');
   282→      } else {
   283→        addToast(data.message || 'Failed to submit application', 'error');
   284→      }
   285→    } catch {
   286→      addToast('Network error. Please try again.', 'error');
   287→    } finally {
   288→      setSubmitting(false);
   289→    }
   290→  };
   291→
   292→  /* ──────── SchoolPay payment ──────── */
   293→
   294→  const handleInitiatePayment = async () => {
   295→    setPaymentError('');
   296→    setPaymentStatus('initiating');
   297→    try {
   298→      const res = await fetch('/api/payments', {
   299→        method: 'POST',
   300→        headers: { 'Content-Type': 'application/json' },
   301→        body: JSON.stringify({
   302→          action: 'initiate',
   303→          referenceNumber,
   304→          fullName: form.fullName,
   305→          phone: form.phone,
   306→          email: form.email,
   307→          amount: applicationFee,
   308→          programme: form.programme,
   309→          intakeYear: form.intakeYear,
   310→        }),
   311→      });
   312→      const data = await res.json();
   313→
   314→      if (data.success) {
   315→        setTransactionRef(data.data.transactionRef);
   316→        setPaymentStatus('dialog');
   317→      } else {
   318→        setPaymentError(data.message || 'Payment initiation failed');
   319→        setPaymentStatus('idle');
   320→      }
   321→    } catch {
   322→      setPaymentError('Network error. Please try again.');
   323→      setPaymentStatus('idle');
   324→    }
   325→  };
   326→
   327→  const handleSimulatePayment = async () => {
   328→    setPaymentStatus('verifying');
   329→    try {
   330→      const res = await fetch('/api/payments', {
   331→        method: 'POST',
   332→        headers: { 'Content-Type': 'application/json' },
   333→        body: JSON.stringify({
   334→          action: 'verify',
   335→          transactionRef,
   336→          referenceNumber,
   337→        }),
   338→      });
   339→      const data = await res.json();
   340→
   341→      if (data.success) {
   342→        setPaymentStatus('success');
   343→        addToast('Payment verified! Admission confirmed.', 'success');
   344→      } else {
   345→        setPaymentStatus('failed');
   346→        setPaymentError(data.message || 'Payment verification failed.');
   347→        addToast(data.message || 'Payment verification failed.', 'error');
   348→      }
   349→    } catch {
   350→      setPaymentStatus('failed');
   351→      setPaymentError('Network error during verification.');
   352→    }
   353→  };
   354→
   355→  /* ──────── document upload ──────── */
   356→
   357→  const MAX_DOC_SIZE = 1.5 * 1024 * 1024; // 1.5MB
   358→
   359→  const handleDocUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
   360→    const file = e.target.files?.[0];
   361→    if (!file) return;
   362→    if (file.size > MAX_DOC_SIZE) {
   363→      addToast(`File too large. Maximum size is 1.5MB. Your file is ${(file.size / 1024 / 1024).toFixed(1)}MB.`, 'error');
   364→      e.target.value = '';
   365→      return;
   366→    }
   367→    setUploadingDoc(true);
   368→    try {
   369→      const formData = new FormData();
   370→      formData.append('file', file);
   371→      const res = await fetch('/api/upload', { method: 'POST', body: formData });
   372→      if (res.ok) {
   373→        const data = await res.json();
   374→        setUploadedDocs((prev) => [...prev, { name: file.name, url: data.url, size: file.size }]);
   375→        addToast(`${file.name} uploaded successfully`, 'success');
   376→      } else {
   377→        addToast('Upload failed. Please try again.', 'error');
   378→      }
   379→    } catch {
   380→      addToast('Network error during upload.', 'error');
   381→    } finally {
   382→      setUploadingDoc(false);
   383→      e.target.value = '';
   384→    }
   385→  };
   386→
