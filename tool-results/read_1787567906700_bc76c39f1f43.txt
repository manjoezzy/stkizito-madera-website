   189→
   190→export default function AdmissionsPage() {
   191→  const { addToast, setCurrentPage } = useAppStore();
   192→
   193→  const [nonFormalFormUrl, setNonFormalFormUrl] = useState<string | null>(null);
   194→  const [tvetFormUrl, setTvetFormUrl] = useState<string>('/forms/tvet-admission-form.pdf');
   195→  const [applicationFee, setApplicationFee] = useState<number>(50000);
   196→  useEffect(() => {
   197→    Promise.all([
   198→      fetch('/api/settings?key=non_formal_form_url').then((r) => r.json()),
   199→      fetch('/api/settings?key=tvet_form_url').then((r) => r.json()),
   200→      fetch('/api/settings?key=application_fee').then((r) => r.json()),
   201→    ]).then(([nf, tf, af]) => {
   202→      if (nf.value) setNonFormalFormUrl(nf.value);
   203→      if (tf.value) setTvetFormUrl(tf.value);
   204→      if (af.value) setApplicationFee(parseInt(af.value, 10) || 50000);
   205→    }).catch(() => {});
   206→  }, []);
   207→
   208→  const [currentStep, setCurrentStep] = useState(1);
   209→  const [form, setForm] = useState<FormData>(INITIAL_FORM);
   210→  const [grades, setGrades] = useState<GradeRow[]>([{ subject: '', grade: '' }]);
   211→  const [errors, setErrors] = useState<Partial<Record<keyof FormData | string, string>>>({});
   212→  const [submitting, setSubmitting] = useState(false);
   213→  const [referenceNumber, setReferenceNumber] = useState('');
   214→  const [schoolpayCode, setSchoolpayCode] = useState('');
   215→  const [uploadedDocs, setUploadedDocs] = useState<UploadedDoc[]>([]);
   216→  const [docErrors, setDocErrors] = useState<Record<string, string>>({});
   217→  const fileInputRefs = useRef<Record<string, HTMLInputElement | null>>({});
   218→
   219→  const updateField = (field: keyof FormData, value: string) => {
   220→    setForm((prev) => ({ ...prev, [field]: value }));
   221→    if (errors[field]) {
   222→      setErrors((prev) => {
   223→        const next = { ...prev };
   224→        delete next[field];
   225→        return next;
   226→      });
   227→    }
   228→  };
   229→
   230→  /* ──────── grades management ──────── */
   231→
   232→  const addGradeRow = () => {
   233→    setGrades((prev) => [...prev, { subject: '', grade: '' }]);
   234→  };
   235→
   236→  const removeGradeRow = (index: number) => {
   237→    setGrades((prev) => prev.filter((_, i) => i !== index));
   238→  };
   239→
   240→  const updateGrade = (index: number, field: 'subject' | 'grade', value: string) => {
   241→    setGrades((prev) => prev.map((row, i) => (i === index ? { ...row, [field]: value } : row)));
   242→  };
   243→
   244→  /* ──────── document management ──────── */
   245→
   246→  const handleFileSelect = (docType: string, label: string, file: File | null) => {
   247→    if (!file) return;
   248→
   249→    if (file.size > 1.5 * 1024 * 1024) {
   250→      addToast(`File too large. Maximum size is 1.5MB. Your file is ${(file.size / 1024 / 1024).toFixed(2)}MB.`, 'error');
   251→      return;
   252→    }
   253→
   254→    const allowedTypes = ['image/jpeg', 'image/png', 'image/webp', 'application/pdf', 'application/msword', 'application/vnd.openxmlformats-officedocument.wordprocessingml.document'];
   255→    if (!allowedTypes.includes(file.type)) {
   256→      addToast('Invalid file type. Upload images, PDFs, or Word documents only.', 'error');
   257→      return;
   258→    }
   259→
   260→    setUploadedDocs((prev) => {
   261→      const filtered = prev.filter((d) => d.key !== docType);
   262→      return [...filtered, { key: docType, label, file, preview: `${file.name} (${(file.size / 1024).toFixed(0)} KB)` }];
   263→    });
   264→    setDocErrors((prev) => {
   265→      const next = { ...prev };
   266→      delete next[docType];
   267→      return next;
   268→    });
   269→  };
   270→
   271→  const removeDoc = (docType: string) => {
   272→    setUploadedDocs((prev) => prev.filter((d) => d.key !== docType));
   273→    if (fileInputRefs.current[docType]) {
   274→      fileInputRefs.current[docType]!.value = '';
   275→    }
   276→  };
   277→
   278→  /* ──────── validation ──────── */
   279→
   280→  const validateStep = (step: number): boolean => {
   281→    const e: Partial<Record<keyof FormData | string, string>> = {};
   282→
   283→    if (step === 1) {
   284→      if (!form.fullName.trim()) e.fullName = 'Full name is required';
   285→      if (!form.dob) e.dob = 'Date of birth is required';
   286→      if (!form.gender) e.gender = 'Gender is required';
   287→      if (!form.nationality.trim()) e.nationality = 'Nationality is required';
   288→    }
   289→
   290→    if (step === 2) {
   291→      if (!form.phone.trim()) e.phone = 'Phone number is required';
   292→      else if (!/^\d{10}$/.test(form.phone.replace(/\s/g, ''))) e.phone = 'Enter a valid 10-digit phone number';
   293→      if (!form.email.trim()) e.email = 'Email is required';
   294→      else if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(form.email)) e.email = 'Enter a valid email address';
   295→    }
   296→
   297→    if (step === 3) {
   298→      if (!form.qualification) e.qualification = 'Select your highest qualification';
   299→      if (!form.institutionLevel) e.institutionLevel = 'Select institution level';
   300→    }
   301→
   302→    if (step === 4) {
   303→      if (!form.programme) e.programme = 'Select a programme';
   304→      if (!form.intakeYear) e.intakeYear = 'Select intake year';
   305→
   306→      // Validate documents
   307→      const de: Record<string, string> = {};
   308→      DOCUMENT_TYPES.forEach((dt) => {
   309→        if (!uploadedDocs.find((d) => d.key === dt.key)) {
   310→          de[dt.key] = `${dt.label} is required`;
   311→        }
   312→      });
   313→      setDocErrors(de);
   314→      if (Object.keys(de).length > 0) {
   315→        e._docs = 'Please upload all required documents';
   316→      }
   317→    }
   318→
   319→    setErrors(e);
   320→    return Object.keys(e).length === 0;
   321→  };
   322→
   323→  const goNext = () => {
   324→    if (!validateStep(currentStep)) return;
   325→    if (currentStep === 4) {
   326→      handleSubmitApplication();
   327→    } else {
   328→      setCurrentStep((s) => Math.min(s + 1, 5));
   329→    }
   330→  };
   331→
   332→  const goPrev = () => {
   333→    setCurrentStep((s) => Math.max(s - 1, 1));
   334→  };
   335→
   336→  /* ──────── submit application (step 4 → step 5) ──────── */
   337→
   338→  const handleSubmitApplication = async () => {
   339→    setSubmitting(true);
   340→    try {
   341→      // 1. Upload all files first
   342→      const uploadedUrls: Array<{ fileName: string; fileUrl: string; fileSize: number; documentType: string }> = [];
   343→
   344→      for (const doc of uploadedDocs) {
   345→        const formData = new FormData();
   346→        formData.append('file', doc.file);
   347→        formData.append('type', 'admission');
   348→
   349→        const uploadRes = await fetch('/api/upload', {
   350→          method: 'POST',
   351→          body: formData,
   352→        });
   353→
   354→        if (!uploadRes.ok) {
   355→          const errData = await uploadRes.json().catch(() => ({ error: 'Upload failed' }));
   356→          throw new Error(errData.error || `Failed to upload ${doc.label}`);
   357→        }
   358→
   359→        const uploadData = await uploadRes.json();
   360→        uploadedUrls.push({
   361→          fileName: uploadData.fileName,
   362→          fileUrl: uploadData.url,
   363→          fileSize: uploadData.fileSize,
   364→          documentType: doc.key,
   365→        });
   366→      }
   367→
   368→      // 2. Submit application with form data + grades + uploaded documents
   369→      const res = await fetch('/api/admissions', {
   370→        method: 'POST',
   371→        headers: { 'Content-Type': 'application/json' },
   372→        body: JSON.stringify({
   373→          ...form,
   374→          grades: grades.filter((g) => g.subject.trim() && g.grade),
   375→          uploadedDocuments: uploadedUrls,
   376→        }),
   377→      });
   378→      const data = await res.json();
   379→
   380→      if (data.success) {
   381→        setReferenceNumber(data.data.referenceNumber);
   382→        setSchoolpayCode(data.data.schoolpayCode || '');
   383→        setCurrentStep(5);
   384→        addToast('Application submitted successfully!', 'success');
   385→      } else {
   386→        addToast(data.message || 'Failed to submit application', 'error');
   387→      }
   388→    } catch (err: unknown) {
   389→      const msg = err instanceof Error ? err.message : 'Network error. Please try again.';
   390→      addToast(msg, 'error');
   391→    } finally {
   392→      setSubmitting(false);
   393→    }
   394→  };

   497→
   498→  /* ──────── step content ──────── */
   499→

   559→              {renderField('nin', 'National Identification Number (NIN)', 'text', 'e.g. CFXXXXXXXXXXXX')}
